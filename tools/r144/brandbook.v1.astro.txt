<script>
  // el índice sigue la lectura: la sección que cruza el tercio superior de la pantalla, a plena tinta
  const enlaces = [...document.querySelectorAll<HTMLAnchorElement>('[data-ir]')];
  const secciones = enlaces.map((a) => document.getElementById(a.dataset.ir!)!).filter(Boolean);
  const marcar = (id: string) => enlaces.forEach((a) => a.toggleAttribute('aria-current', a.dataset.ir === id));
  const io = new IntersectionObserver((es) => { for (const e of es) if (e.isIntersecting) marcar(e.target.id); }, { rootMargin: '-30% 0px -65% 0px' });
  secciones.forEach((s) => io.observe(s));
</script>
