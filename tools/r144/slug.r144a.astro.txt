//   nombre · entradilla y ficha (las cifras, pequeñas, en la ficha) · portada · la frase · COLOR (bandas de su paleta que crecen al llegar) ·
//   un DETALLE de la web junto al reto · galería · TIPOGRAFÍA (su muestra, escrita en SU web con SU fuente, que se imprime al llegar) · la web sección
//   a sección · galería · apartados con otro DETALLE · galería · por dentro y lo descartado · la cita · el siguiente.

import Base from '../../layouts/Base.astro';
import Nav from '../../components/Nav.astro';
import Carriles from '../../components/Carriles.astro';
import Pie from '../../components/Pie.astro';
import Cta from '../../components/Cta.astro';
import Medio from '../../components/obra/Medio.astro';
import { proyectos, type Proyecto, type Medio as TMedio } from '../../data/proyectos';

export function getStaticPaths() {
  return proyectos.map((p, i) => ({ params: { slug: p.slug }, props: { p, sig: proyectos[(i + 1) % proyectos.length] } }));
}
const { p, sig } = Astro.props as { p: Proyecto; sig: Proyecto };
const g = [...p.galeria];
const tomar = (): TMedio[] => {
  if (!g.length) return [];
  if (g[0].ancho === 'mitad' && g[1]?.ancho === 'mitad') return g.splice(0, 2);
  return g.splice(0, 1);
};
const huecos = [tomar(), tomar(), g.splice(0)];
const det = (n: number): TMedio => ({ tipo: 'video', src: `/media/obra/${p.slug}/d${n}.mp4`, poster: `/media/obra/${p.slug}/d${n}.jpg` });
const ficha = [
  ['Servicios', p.servicios],
  ['Sector', p.sector ? [p.sector] : []],
  ['Lugar', p.ubicacion ? [p.ubicacion] : []],
  ['Con', p.con ? [p.con] : []],
  ['Herramientas', p.stack ?? []],
  ['En cifras', (p.cifras ?? []).map((c) => `${c.v} ${c.t}`)],
].filter(([, v]) => (v as string[]).length) as [string, string[]][];
const luz = (hex: string) => { const n = parseInt(hex.slice(1), 16), r = n >> 16, gg = (n >> 8) & 255, b = n & 255; return (r * 299 + gg * 587 + b * 114) / 1000 > 150; };
// el fondo de las muestras tipográficas: el color más claro de su paleta (la tinta de la muestra es la suya)
const claroMarca = (p.paleta ?? []).map((c) => c.hex).find((h) => luz(h)) ?? '#FFFFFF';
const media = (m: TMedio, dos: boolean) => (dos ? (m.movil ? '4 / 5' : '4 / 3') : m.movil ? '16 / 10' : '16 / 9');
import { flecha } from '../../lib/flecha';
