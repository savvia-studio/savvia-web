    let mx = 0, my = 0, tx = 0, ty = 0, rafC = 0;
    const mover = () => {
      rafC = 0;
      const r = pie.getBoundingClientRect(), p = Math.min(1, Math.max(0, 1 - r.top / innerHeight)); // 0 al asomar, 1 con el pie arriba
      mx += (tx - mx) * 0.08; my += (ty - my) * 0.08;
      capasEl.forEach((c) => { const k = +c.dataset.prof!; c.style.transform = `translate3d(${(mx * 22 * k).toFixed(1)}px, ${((1 - p) * 90 * k + my * 12 * k).toFixed(1)}px, 0)`; });
      if (Math.abs(tx - mx) + Math.abs(ty - my) > 0.002) rafC = requestAnimationFrame(mover);
    };
    if (!reducido) {
      addEventListener('scroll', () => { if (!rafC) rafC = requestAnimationFrame(mover); }, { passive: true });
      pie.addEventListener('pointermove', (e) => { tx = e.clientX / innerWidth - 0.5; ty = e.clientY / innerHeight - 0.5; if (!rafC) rafC = requestAnimationFrame(mover); }, { passive: true });
      mover();
    }
    const pintarSala = () => td?.soltar(); // ronda 94: el director se compone desde la batuta al llegar
    new IntersectionObserver((e, io) => { if (e[0].isIntersecting) { montarDirector(); pintarSala(); llegar(); io.disconnect(); } }, { threshold: 0.3 }).observe(final);
    document.fonts.ready.then(medir); addEventListener('resize', medir); medir();
  }
</script>
