<script>
  import { EST, estacionActual, fijarEstacion } from '../scripts/estacion';
  import { sonidoEncendido, fijarSonido } from '../scripts/sonidos';
  const caja = document.querySelector<HTMLElement>('[data-mando-flot]');
  if (caja) {
    const mas = caja.querySelector<HTMLButtonElement>('[data-mando-mas]')!, grupo = caja.querySelector<HTMLElement>('.grupo')!;
    const ests = [...caja.querySelectorAll<HTMLButtonElement>('[data-mando-est]')], son = caja.querySelector<HTMLButtonElement>('[data-mando-son]')!;
    const reducido = matchMedia('(prefers-reduced-motion: reduce)').matches;
    let abierto = false, finGoo = 0;
    const abrir = (si: boolean) => {
      if (si === abierto) return;
      abierto = si;
      caja.classList.toggle('abierto', si);
      mas.setAttribute('aria-expanded', String(si));
      grupo.inert = !si;
      if (!reducido) { caja.classList.add('fundiendo'); clearTimeout(finGoo); finGoo = window.setTimeout(() => caja.classList.remove('fundiendo'), si ? 900 : 650); }
    };
    mas.addEventListener('click', () => abrir(!abierto));
    document.addEventListener('pointerdown', (e) => { if (abierto && !caja.contains(e.target as Node)) abrir(false); });
    addEventListener('keydown', (e) => { if (e.key === 'Escape' && abierto) abrir(false); });
    const pintarEst = () => { const n = EST[estacionActual()]; ests.forEach((b) => b.setAttribute('aria-pressed', String(b.dataset.mandoEst === n))); };
    ests.forEach((b) => b.addEventListener('click', () => fijarEstacion(EST.indexOf(b.dataset.mandoEst as (typeof EST)[number]), 'mando')));
    addEventListener('savvia:estacion', pintarEst);
    pintarEst();
    const pintarSon = (on: boolean) => { son.setAttribute('aria-pressed', String(on)); son.setAttribute('aria-label', on ? 'Sonido activado. Silenciar' : 'Sonido silenciado. Activar'); };
    pintarSon(sonidoEncendido());
    son.addEventListener('click', () => fijarSonido(!sonidoEncendido()));
    addEventListener('savvia:son', (e) => pintarSon((e as CustomEvent).detail.on));
  }
</script>
