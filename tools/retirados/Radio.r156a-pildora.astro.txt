<script>
  const caja = document.querySelector<HTMLElement>('[data-radio]');
  if (caja) {
    type Emisora = { id: string; nombre: string; pistas: { f: string; t: string }[] };
    const EM = JSON.parse(caja.dataset.emisoras!) as Emisora[];
    const bPlay = caja.querySelector<HTMLButtonElement>('[data-radio-play]')!, bEm = caja.querySelector<HTMLButtonElement>('[data-radio-emisora]')!;
    const nombre = caja.querySelector<HTMLElement>('[data-radio-nombre]')!, pista = caja.querySelector<HTMLElement>('[data-radio-pista]')!;
    const a = new Audio(); a.preload = 'none'; a.volume = 0.5;
    let em = 0, idx = -1, on = false, token = 0;
    try { const g = localStorage.getItem('savvia:radio'); const k = EM.findIndex((e) => e.id === g); if (k >= 0) em = k; } catch {}
    const pintar = () => {
      caja.classList.toggle('suena', on);
      nombre.textContent = EM[em].nombre;
      pista.textContent = on && idx >= 0 ? EM[em].pistas[idx].t : '';
      bPlay.setAttribute('aria-label', on ? 'Apagar la radio' : 'Encender la radio');
      bEm.setAttribute('aria-label', `Emisora: ${EM[em].nombre}. Cambiar a ${EM[(em + 1) % EM.length].nombre}`);
    };
    const avisar = () => dispatchEvent(new CustomEvent('savvia:radio', { detail: { on } }));
    const otra = () => { const n = EM[em].pistas.length; let k = idx; while (n > 1 && k === idx) k = Math.floor(Math.random() * n); if (n === 1) k = 0; idx = k; };
    // una pieza al azar; `aMedias` = entrar a mitad de ella (al encender o al cambiar de emisora, como una radio); al encadenar, desde el principio
    const sonar = async (aMedias: boolean) => {
      const t = ++token; otra();
      a.src = `/media/radio/${EM[em].id}/${EM[em].pistas[idx].f}.mp3`;
      a.load();
      await new Promise<void>((r) => { if (a.readyState >= 1) r(); else a.addEventListener('loadedmetadata', () => r(), { once: true }); });
      if (t !== token) return;
      if (aMedias && Number.isFinite(a.duration) && a.duration > 60) a.currentTime = Math.random() * (a.duration - 45);
      try { await a.play(); } catch { if (t === token) { on = false; pintar(); avisar(); } return; }
      if (t === token) pintar();
    };
    a.addEventListener('ended', () => { if (on) sonar(false); });
    a.addEventListener('error', () => { if (on) sonar(false); }); // una pieza que no carga: la siguiente
    const encender = (si: boolean) => {
      on = si;
      try { sessionStorage.setItem('savvia:radio-on', si ? '1' : '0'); } catch {}
      if (si) sonar(true); else { token++; a.pause(); pintar(); }
      avisar();
    };
    bPlay.addEventListener('click', () => encender(!on));
    bEm.addEventListener('click', () => {
      em = (em + 1) % EM.length; idx = -1;
      try { localStorage.setItem('savvia:radio', EM[em].id); } catch {}
      if (on) sonar(true); else pintar();
    });
    pintar();
    // sigue sonando al cambiar de página (misma sesión); si el navegador no deja arrancar sin un gesto, se queda apagada
    try { if (sessionStorage.getItem('savvia:radio-on') === '1') encender(true); } catch {}
    (window as any).__radio = () => ({ on, emisora: EM[em].id, pista: idx >= 0 ? EM[em].pistas[idx].f : null, t: a.currentTime, pausado: a.paused, vol: a.volume });
  }
</script>
