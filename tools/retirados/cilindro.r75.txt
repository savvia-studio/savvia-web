<script>
  // RONDA 53 · EL CILINDRO. Andoni: «la sección de más proyectos: al hacer scroll, que parezca como que la página está curvada hacia dentro, como si fuese
  // un cilindro». Cada bloque (cabecera, piezas, voces) se coloca sobre la cara INTERIOR de un cilindro horizontal de radio R centrado en la pantalla:
  // a un ángulo φ = d / R de la mitad de la pantalla, se acerca R·(1 − cos φ) (translateZ), se recoge hacia el centro (R·sen φ − d) y se inclina φ para mirar
  // a quien lee (rotateX: por debajo del centro, su borde de abajo se acerca; por encima, el de arriba). El centro de cada bloque se mide SIN transformar
  // (al cargar, al cambiar de tamaño la sección y al redimensionar) y el scroll solo escribe `transform`. Solo con la sección a la vista; sin movimiento reducido.
  {
    const sec = document.getElementById('anteriores');
    if (sec && !matchMedia('(prefers-reduced-motion: reduce)').matches) {
      const els = [...sec.querySelectorAll<HTMLElement>('.cabecera, .mosaico > .pieza, .mosaico > .banda')];
      let centros: number[] = [], vista = false, raf = 0;
      const medir = () => {
        els.forEach((e) => (e.style.transform = ''));
        centros = els.map((e) => { const r = e.getBoundingClientRect(); return r.top + scrollY + r.height / 2; });
        pintar();
      };
      const pintar = () => {
        raf = 0;
        const vh = innerHeight, R = vh * (innerWidth < 768 ? 1.25 : 0.95), P = vh * 1.6;
        els.forEach((e, k) => {
          const d = centros[k] - scrollY - vh / 2;
          const dc = Math.max(-1.25 * R, Math.min(1.25 * R, d)), phi = dc / R; // más allá del tope, el bloque se queda como en el tope (sin volver a entrar)
          const z = R * (1 - Math.cos(phi)) * 0.55, dy = (R * Math.sin(phi) - dc) * 0.55, rx = (phi * 180) / Math.PI * 0.62;
          e.style.transform = `perspective(${P.toFixed(0)}px) translate3d(0, ${dy.toFixed(1)}px, ${z.toFixed(1)}px) rotateX(${rx.toFixed(2)}deg)`;
        });
      };
      const pedir = () => { if (vista && !raf) raf = requestAnimationFrame(pintar); };
      new IntersectionObserver((es) => { vista = es[0].isIntersecting; pedir(); }, { rootMargin: '50% 0px' }).observe(sec);
      addEventListener('scroll', pedir, { passive: true });
      addEventListener('resize', medir);
      new ResizeObserver(() => medir()).observe(sec);
      addEventListener('load', medir);
      document.fonts?.ready.then(medir);
      (window as any).__cilindro = () => els.map((e) => e.style.transform);
    }
  }
</script>
