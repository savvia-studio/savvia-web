<script>
  // RONDA 54–55 · LA VENTANA. Una ventanita con contraventanas de lamas bajo el subtítulo. Ronda 55: «cuando se va la luz no está bien sincronizado y no
  // parece realista»: la luz no «viaja»; sale de la ABERTURA. Un solo reloj (`ap`, 0…1) gira las hojas Y recorta el haz a la rendija que dejan entre

  // la parte abierta de los cuatro cristales a lo largo de la dirección de la luz (paralela), abriéndose hacia el suelo; motas de polvo dentro.

      const hojas = [...btn.querySelectorAll<HTMLElement>('[data-hoja]')];
      const marco = btn.querySelector<HTMLElement>('.marco')!;
      const NS = 'http://www.w3.org/2000/svg';
      let abierta = false, ap = 0, raf = 0;
      const polys = Array.from({ length: 4 }, () => { const p = document.createElementNS(NS, 'polygon'); p.setAttribute('fill', 'url(#luz-v)'); panes.append(p); return p; });
      const puntos = Array.from({ length: 14 }, (_, n) => { const c = document.createElementNS(NS, 'circle'); c.setAttribute('r', String(0.8 + (n % 3) * 0.5)); c.style.setProperty('--n', String(n)); motas.append(c); return c; });
      // RONDA 56 · LA LUZ QUE SE FILTRA ENTRE LAS LAMAS: con las hojas cerradas, cada rendija deja pasar una franja fina de luz que se proyecta (y se abre)
      // hacia el suelo; al abrir, las lamas giran con la hoja y las franjas se apagan (la luz pasa ya por la abertura)
      const fugas = svg.querySelector<SVGGElement>('[data-luz-fugas]')!, gradF = svg.querySelector<SVGLinearGradientElement>('[data-luz-grad-f]')!;
      const rayas = svg.querySelector<SVGPatternElement>('[data-luz-rayas]')!, mfuga = svg.querySelector<SVGGElement>('[data-luz-fuga-m]')!;
      const tiras = Array.from({ length: 4 }, () => { const p = document.createElementNS(NS, 'polygon'); p.setAttribute('fill', 'url(#luz-rayas)'); fugas.append(p); return p; });

      // LA PALABRA ILUMINADA: una copia de «Contáctanos» en luz cálida, recortada a la forma del haz (y de las franjas), con su SOMBRA desplazada en la
      // dirección de la luz; entre los árboles y la hierba, como la palabra
      const palabra = pie.querySelector<HTMLElement>('[data-contacta-txt]');
      const paisaje = pie.querySelector<HTMLElement>('[data-paisaje]');
      const lit = document.createElement('span');
      lit.className = 'contacta-luz'; lit.setAttribute('aria-hidden', 'true'); lit.textContent = palabra?.textContent ?? '';
      paisaje?.append(lit);
      const sombras = svg.querySelector<SVGGElement>('[data-luz-siluetas]')!;
      const PASOS_S = 90;
      const usos = Array.from({ length: PASOS_S }, () => { const u = document.createElementNS(NS, 'use'); u.setAttribute('href', '#siluetas-v'); sombras.append(u); return u; });
      const quad = (P: number[][]) => P.map(([a, b]) => `${a.toFixed(1)},${b.toFixed(1)}`).join(' ');

        hojas.forEach((h) => (h.style.rotate = `y ${(Number(h.dataset.hoja) * ap * 118).toFixed(1)}deg`));

        const fuga = Math.max(0, 1 - s * 1.6);
        const cristales = [[x0, cx - g / 2], [cx + g / 2, x0 + w]].flatMap(([l, r]) => [y0, y0 + hh + g].map((py) => [[l, py], [r, py], lejos(r, py), lejos(r, py + hh), lejos(l, py + hh), [l, py + hh]]));

        }
      };
      const ir = (meta: number, dur: number) => {
        cancelAnimationFrame(raf);
        if (reducido) { ap = meta; dibujar(); return; }
        const a0 = ap, t0 = performance.now();
        const paso = (ahora: number) => {
          const p = Math.min(1, (ahora - t0) / dur), e = p < 0.5 ? 4 * p * p * p : 1 - Math.pow(-2 * p + 2, 3) / 2; // la madera: arranca y frena

        raf = requestAnimationFrame(paso);

    const bucle = (ahora: number) => { rafC = 0; if (!visC) return; dibujarC(ahora); if (!reducido || fadeMeta !== fade) rafC = requestAnimationFrame(bucle); };
    const arrancar = () => { if (!rafC && visC) { t0C = 0; rafC = requestAnimationFrame(bucle); } };
    new ResizeObserver(() => { medir(); if (!pzs.length) sembrar(); arrancar(); }).observe(pie);
    new IntersectionObserver((e) => { visC = e[0].isIntersecting; arrancar(); }).observe(pie);

    // ── LAS FLORES DE PRIMAVERA (ronda 44 · «flores que encajen con el fondo; lo de ahora es un pegote»). Van en un lienzo DENTRO de la capa de la
    // hierba: nacen sobre el prado real (su borde de arriba, HORIZONTE, en coordenadas de la propia lámina) y se mueven con él (paralaje y ratón).
    // Son florecillas PINTADAS con el toque del prado: manchas blandas, blanco y rosa pálido (algo de Rosa flor), en MATAS, más pequeñas cuanto más
    // lejos, que BROTAN poco a poco cuando llegas al pie. Solo en primavera.
    const cf = pie.querySelector<HTMLCanvasElement>('[data-flores]')!, fx = cf.getContext('2d')!;
    cf.classList.add('listo');
    type Fl = { x: number; d: number; z: number; v: number; c: string };
    let flores: Fl[] = [], nacF = 0, rafF = 0, dprF = 1, WF = 0, HF = 0;
    const sembrarF = () => {
      const centros = Array.from({ length: 60 }, () => [Math.random(), Math.pow(Math.random(), 1.4)]); // x, profundidad bajo el borde del prado (0 = lejos)
      flores = Array.from({ length: 280 }, (_, i) => { const [cx0, cd] = centros[i % centros.length]; const d = Math.min(1, Math.max(0, cd + (Math.random() - 0.5) * 0.05)); return { x: cx0 + (Math.random() - 0.5) * (0.012 + d * 0.02), d, z: Math.random(), v: Math.random(), c: FLORES[Math.floor(Math.random() * FLORES.length)] }; });
    };
    const bordeF = (x: number) => { const i = Math.min(1, Math.max(0, x)) * (HZ.length - 1), a0 = Math.floor(i), a1 = Math.min(HZ.length - 1, a0 + 1); return HZ[a0] + (HZ[a1] - HZ[a0]) * (i - a0); };
    const pintarF = (ahora: number) => {
      rafF = 0;
      const r = cf.getBoundingClientRect(), d2 = Math.min(devicePixelRatio || 1, 1.5);
      if (Math.round(r.width * d2) !== cf.width || Math.round(r.height * d2) !== cf.height) { cf.width = Math.round(r.width * d2); cf.height = Math.round(r.height * d2); }
      dprF = d2; WF = r.width; HF = r.height;
      fx.setTransform(dprF, 0, 0, dprF, 0, 0); fx.clearRect(0, 0, WF, HF);
      if (est !== 2) return;
      const edad = (ahora - nacF) / 1000; let pendientes = false;
      for (const f of flores) {
        const abre = reducido ? 1 : Math.max(0, Math.min(1, (edad - f.v * 7) / 1.1));
        if (abre < 1) pendientes = true;
        if (!abre) continue;
        const top = bordeF(f.x) + 0.015, y = (top + f.d * (0.97 - top)) * HF;
        const rr = (1.3 + f.d * 3.4) * (0.75 + f.z * 0.5) * (1 - Math.pow(1 - abre, 3)) * (WF / 1280);
        florecilla(fx, f.x * WF, y, Math.max(0.6, rr), f.c, 0.7 + f.d * 0.28);
      }
      if (pendientes && visC) rafF = requestAnimationFrame(pintarF);
    };
    const pedirF = () => { if (!rafF) rafF = requestAnimationFrame(pintarF); };
    let vistoF = false;
    new IntersectionObserver((e) => { if (e[0].isIntersecting && !vistoF && paisaje.classList.contains('pintado')) { vistoF = true; nacF = performance.now(); } pedirF(); }, { threshold: 0.3 }).observe(pie);
    new MutationObserver(() => { if (paisaje.classList.contains('pintado') && !vistoF) { vistoF = true; nacF = performance.now() + 400; pedirF(); } }).observe(paisaje, { attributes: true, attributeFilter: ['class'] });
    new ResizeObserver(pedirF).observe(cf);
    addEventListener('savvia:estacion', () => { setTimeout(() => { nacF = performance.now(); pedirF(); }, 1200); });
    sembrarF();
    addEventListener('savvia:estacion', () => { fadeMeta = 0; arrancar(); }); // se apaga lo de antes; al apagarse, entra la estación nueva (est)
    if (reducido) { medir(); sembrar(); }
  }
</script>
