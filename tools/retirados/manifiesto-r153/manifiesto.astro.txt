// RONDA 126 · EL MANIFIESTO. RONDA 128: en claro, tipo clusia/about. RONDAS 129–143: en papel (roble gofrado, fibras, «la imprenta»).
// RONDA 144 · Eric: «mejor como 3–4 párrafos seguidos»: sin capítulos ni numerales romanos.
// RONDA 153 · Andoni: «Manifesto tenemos que darle una vuelta, quitarle textura e igual tirar por algo más natural, no tanto papel.»
// Fuera TODO el papel (fibras, grano, gofrado, imprenta). La idea, una: TODO EN ESTA PÁGINA SE PINTA COMO EL ROBLE.
// · Portada: «Manifiesto» y, debajo, el roble de la estación de hoy en acuarela (el motor del hero, estela.ts): cae una gota de savia, el árbol se
//   dibuja a línea y se pinta con el canto rasgado; después sus raíces bajan desde el cuello. Quieto en la página (no se mueve con el scroll ni con el ratón).
// · El texto: tres párrafos seguidos que crecen de la raíz. Cada uno se PINTA al llegar con el mismo canto rasgado del árbol (máscara que se abre desde
//   su primera palabra) y su frase subrayada recibe la savia después. Ese es el paso de un párrafo a otro.
// · La estación manda (html[data-estacion]): fondo, tinta, savia y lámina. Si cambia, el roble nuevo se pinta encima del viejo desde la base.
// · Movimiento reducido o sin JS: todo pintado y quieto.
// RONDA 153 · Andoni: «El manifiesto en realidad es nuestra forma de ser, de actuar: lo que fluye por dentro del árbol. El texto del medio necesita
// mucho más aire y empezar con una letra estilo pluma.» → EL MANIFIESTO ES LA SAVIA (components/manifiesto/savia.ts): un hilo de savia baja por dentro
// de la raíz, cae a la capitular y la ESCRIBE (la T de pluma de punta ancha, capitular.ts), sale por su cola a un canal junto a la columna y alimenta
// cada párrafo al llegar a su altura (la tinta sube fila a fila); pulsos de luz suben hacia el árbol mientras lees. Fuera el canto rasgado sobre el
// texto (se leía como un corte vertical).

// [[…]] = la frase que se subraya con savia
const rico = (t: string) =>
  t.split(/(\[\[.+?\]\])/).filter(Boolean).map((seg) => {
    const m = /^\[\[(.+)\]\]$/.exec(seg);
    return m ? { t: m[1], m: true } : { t: seg, m: false };
  });
const parrafos = manifiesto.capitulos.map((c) => rico(c.parrafos.join(' ')));
// la primera letra va a la capitular; el párrafo empieza en la segunda
const inicial = parrafos[0][0].t.charAt(0);
parrafos[0][0] = { ...parrafos[0][0], t: parrafos[0][0].t.slice(1) };

<script>
  import { crearEstela, ESTACIONES_SIN_RALEO, type Estela } from '../scripts/estela';
  import { EST, estacionActual } from '../scripts/estacion';
  import { rgbDe } from '../scripts/tinta';
  import { iniciarSavia } from '../components/manifiesto/savia';

  const main = document.querySelector<HTMLElement>('[data-manifiesto]')!;
  const roble = main.querySelector<HTMLElement>('[data-roble]')!;
  const imgCopa = main.querySelector<HTMLImageElement>('[data-copa-img]')!;
  const imgRaiz = main.querySelector<HTMLImageElement>('[data-raiz-img]')!;
  let cvCopa = main.querySelector<HTMLCanvasElement>('[data-copa]')!;
  let cvRaiz = main.querySelector<HTMLCanvasElement>('[data-raiz]')!;
  const reducido = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const RASGO = "url('/media/rasgo-circulo.png')";

  // ── LA MÁSCARA RASGADA: el canto de papel de acuarela (el mismo con que el hero repinta el árbol) se abre desde (ox, oy) ──
  const pon = (el: HTMLElement, m: string) => { el.style.setProperty('-webkit-mask', m); el.style.setProperty('mask', m); };
  const revelar = (el: HTMLElement, ox: number, oy: number, dur: number, fin?: () => void, viejo?: HTMLElement) => {
    let a = 0;
    const paso = (ahora: number) => {
      if (!a) a = ahora;
      const k = Math.min(1, (ahora - a) / dur), e = k * k * (3 - 2 * k), w = el.offsetWidth, h = el.offsetHeight;
      const S = Math.max(8, Math.hypot(w, h) * 2.4 * e);
      pon(el, `${RASGO} ${(ox * w - S / 2).toFixed(1)}px ${(oy * h - S / 2).toFixed(1)}px / ${S.toFixed(1)}px ${S.toFixed(1)}px no-repeat`);
      if (viejo) viejo.style.opacity = String(1 - Math.min(1, Math.max(0, (k - 0.15) / 0.75)));
      if (k < 1) requestAnimationFrame(paso);
      else { pon(el, ''); fin?.(); }
    };
    pon(el, `${RASGO} 0 0 / 0 0 no-repeat`);
    requestAnimationFrame(paso);
  };

  // ── EL ROBLE: la lámina de la estación, pintada por el motor del hero ──
  // copa: gota de savia, líneas, pintura rasgada desde la base. En invierno no hierve (ronda 144: «más blanquecino»); en primavera el motor tiñe (cerezo).
  const tinteCopa = (n: number) => (n === 2 ? 0.9 : 0);
  const TINTE_RAIZ = [0, 0.8, 0.65, 0.5]; // otoño · invierno · primavera · verano (Raices.astro, ronda 81)
  const fondo = () => rgbDe('niebla', main); // el claro de la página: el árbol se funde hacia él
  const OPC_COPA = (n: number, gota: boolean) => ({ origen: [0.5, 1] as [number, number], dir: -1 as const, soplo: n !== 1, gota, manual: gota, hd: true, sinLuz: true, intensidad: tinteCopa(n) });
  const OPC_RAIZ = (n: number) => ({ origen: [0.5, 0] as [number, number], dir: 1 as const, soplo: false, sinLuz: true, estaciones: ESTACIONES_SIN_RALEO, intensidad: TINTE_RAIZ[n] });
  const cargar = (src: string) => { const i = new Image(); i.decoding = 'async'; i.src = src; return i; };
  const soltarLienzo = (c: HTMLCanvasElement) => {
    const g = c.getContext('webgl2') || c.getContext('webgl');
    if (g && !g.isContextLost()) g.getExtension('WEBGL_lose_context')?.loseContext();
    c.remove();
  };

  let est = estacionActual();
  imgCopa.src = `/media/hero/arbol-${EST[est]}.webp`;
  imgRaiz.src = `/media/hero/raices-${EST[est]}.webp`;
  const laminaCopa = cargar(imgCopa.src), laminaRaiz = cargar(imgRaiz.src);
  let copa: Estela | null = crearEstela(cvCopa, laminaCopa, OPC_COPA(est, true));
  let raiz: Estela | null = copa ? crearEstela(cvRaiz, laminaRaiz, OPC_RAIZ(est)) : null;
  let visible = true, gRaiz = reducido ? 1 : 0, creciendo = false;

  if (copa && raiz) {
    roble.classList.add('gl');
    if (est === 1 || est === 2) copa.estacion(est);
    raiz.estacion(est);
    copa.fondo(...fondo()); raiz.fondo(...fondo());
    raiz.crecer(gRaiz);
    // las raíces bajan desde el cuello cuando la copa ya ha brotado (3,6 s, la curva del árbol)
    const crecerRaiz = () => {
      if (creciendo || gRaiz >= 1) return;
      creciendo = true;
      let a = 0;
      const paso = (ahora: number) => {
        if (!a) a = ahora;
        const k = Math.min(1, (ahora - a) / 3600);
        gRaiz = 1 - Math.pow(1 - k, 1.6);
        raiz?.crecer(gRaiz);
        if (k < 1) requestAnimationFrame(paso);
      };
      requestAnimationFrame(paso);
    };
    Promise.all([laminaCopa.decode(), laminaRaiz.decode()]).catch(() => null).then(() => {
      setTimeout(() => {
        copa?.soltar();
        if (reducido) return;
        // abierta ya más abajo: raíces pintadas sin más
        if (roble.getBoundingClientRect().bottom < 0) { gRaiz = 1; raiz?.crecer(1); return; }
        setTimeout(crecerRaiz, 2600);
      }, reducido ? 0 : 350);
    });
    // fuera de la pantalla, el roble no se pinta
    new IntersectionObserver((e) => { visible = e[0].isIntersecting; copa?.visible(visible); raiz?.visible(visible); }, { rootMargin: '10% 0px' }).observe(roble);

    // ── CAMBIO DE ESTACIÓN: la lámina nueva viene pintada y se descubre con el canto rasgado desde la base; las raíces, 0,7 s después ──
    let pedida = -1;
    addEventListener('savvia:estacion', (ev) => {
      const n = (ev as CustomEvent).detail.n as number;
      if (n === est) return;
      est = n; pedida = n;
      const nc = cargar(`/media/hero/arbol-${EST[n]}.webp`), nr = cargar(`/media/hero/raices-${EST[n]}.webp`);
      Promise.all([nc.decode(), nr.decode()]).catch(() => null).then(() => {
        if (pedida !== n) return;
        imgCopa.src = nc.src; imgRaiz.src = nr.src;
        const c2 = cvCopa.cloneNode(false) as HTMLCanvasElement, r2 = cvRaiz.cloneNode(false) as HTMLCanvasElement;
        cvCopa.after(c2); cvRaiz.after(r2);
        const tc = crearEstela(c2, nc, OPC_COPA(n, false)), tr = crearEstela(r2, nr, OPC_RAIZ(n));
        if (!tc || !tr) { c2.remove(); r2.remove(); return; }
        if (n === 1 || n === 2) tc.estacion(n);
        tr.estacion(n);
        tc.fondo(...fondo()); tr.fondo(...fondo());
        tc.crecer(1); tr.crecer(gRaiz);
        tc.visible(visible); tr.visible(visible);
        const vc = cvCopa, vr = cvRaiz;
        cvCopa = c2; cvRaiz = r2; copa = tc; raiz = tr;
        if (!visible || reducido) { soltarLienzo(vc); soltarLienzo(vr); return; }
        revelar(c2, 0.5, 1, 2600, () => soltarLienzo(vc), vc);
        setTimeout(() => revelar(r2, 0.5, 0, 2200, () => soltarLienzo(vr), vr), 700);
      });
    });
  }

  // ── LA SAVIA: espera a que la raíz haya bajado (o a que no haya motor) ──
  const savia = iniciarSavia(main, { reducido, listo: () => !copa || gRaiz >= 0.85 || roble.getBoundingClientRect().bottom < 0 });

  (window as any).__manifiesto = () => ({
    roble: roble.getBoundingClientRect().toJSON(),
    gl: roble.classList.contains('gl'),
    raiz: gRaiz,
    savia: savia.sonda(),
  });
</script>
