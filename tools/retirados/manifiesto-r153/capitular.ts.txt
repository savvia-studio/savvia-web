// RONDA 153 · LA CAPITULAR «ESTILO PLUMA»: la T del manifiesto escrita con una pluma de punta ancha (no es una fuente: es el trazo de una pluma).
// Cada trazo es una línea central (curvas de Bézier) por la que pasa una punta de 26–28 unidades sostenida a 32°: la tinta es la unión de los cuadriláteros
// que barre esa punta entre muestra y muestra (un solo <path>, todos en el mismo sentido: con nonzero se suman sin costuras). Por eso las verticales salen
// gruesas, la barra media y las diagonales a 32° finas, como con una pluma de verdad. Se calcula al compilar (lo importa el frontmatter).
// Las líneas centrales se devuelven también: con ellas se ESCRIBE la letra (máscara con stroke-dashoffset, como la «v» de la carga).

type Pt = [number, number];
type Trazo = { curvas: number[]; ancho: number };

export const CAJA = { w: 200, h: 232 };
const ANG = (32 * Math.PI) / 180;
const NIB: Pt = [Math.cos(ANG), -Math.sin(ANG)];

// los tres trazos de tinta (ronda 153, 2.ª: «se leía como una J»): barra ancha centrada, asta recta y un remate corto abajo
const TRAZOS: Trazo[] = [
  // la barra: ancha y centrada sobre el asta, con una leve ondulación y los extremos que caen un poco (remates)
  { curvas: [8, 50, 26, 36, 66, 44, 100, 40, 134, 36, 172, 32, 192, 46], ancho: 26 },
  // el asta: recta y vertical, en el eje
  { curvas: [100, 44, 98.5, 100, 101.5, 160, 100, 214], ancho: 28 },
  // el remate de abajo: un pie corto y horizontal, centrado
  { curvas: [84, 221, 92, 215, 108, 215, 118, 220], ancho: 16 },
];
// la cola: un perfil de savia que sale del remate y se va por abajo a la izquierda, hacia el canal (allí sigue la savia)
const COLA = [118, 220, 124, 226, 104, 230, 74, 230, 44, 230, 20, 230, 4, 231];

const bez = (p: number[], i: number, t: number): Pt => {
  const u = 1 - t;
  const a = u * u * u, b = 3 * u * u * t, c = 3 * u * t * t, d = t * t * t;
  return [a * p[i] + b * p[i + 2] + c * p[i + 4] + d * p[i + 6], a * p[i + 1] + b * p[i + 3] + c * p[i + 5] + d * p[i + 7]];
};
const muestras = (p: number[], porTramo = 64): Pt[] => {
  const out: Pt[] = [];
  for (let i = 0; i + 7 < p.length; i += 6) for (let k = i ? 1 : 0; k <= porTramo; k++) out.push(bez(p, i, k / porTramo));
  return out;
};
const dCurva = (p: number[]) => {
  let d = `M${p[0]} ${p[1]}`;
  for (let i = 2; i + 5 < p.length; i += 6) d += `C${p.slice(i, i + 6).join(' ')}`;
  return d;
};
const f = (n: number) => n.toFixed(2);

const cinta = (t: Trazo) => {
  const pts = muestras(t.curvas);
  const n = pts.length - 1;
  let d = '';
  for (let i = 0; i < n; i++) {
    // presión: la punta entra y sale algo más estrecha (0,55 → 1 → 0,7)
    const pr = (k: number) => { const x = k / n; return x < 0.18 ? 0.55 + 0.45 * Math.sin((x / 0.18) * Math.PI / 2) : x > 0.86 ? 0.7 + 0.3 * Math.cos(((x - 0.86) / 0.14) * Math.PI / 2) : 1; };
    const h0 = (t.ancho / 2) * pr(i), h1 = (t.ancho / 2) * pr(i + 1);
    const [a, b] = [pts[i], pts[i + 1]];
    let q: Pt[] = [
      [a[0] + NIB[0] * h0, a[1] + NIB[1] * h0], [b[0] + NIB[0] * h1, b[1] + NIB[1] * h1],
      [b[0] - NIB[0] * h1, b[1] - NIB[1] * h1], [a[0] - NIB[0] * h0, a[1] - NIB[1] * h0],
    ];
    let area = 0;
    for (let k = 0; k < 4; k++) { const [x0, y0] = q[k], [x1, y1] = q[(k + 1) % 4]; area += x0 * y1 - x1 * y0; }
    if (area < 0) q = q.reverse();
    // un pelo de solape para que no quede costura entre cuadriláteros
    d += `M${f(q[0][0])} ${f(q[0][1])}L${f(q[1][0])} ${f(q[1][1])}L${f(q[2][0])} ${f(q[2][1])}L${f(q[3][0])} ${f(q[3][1])}Z`;
  }
  return d;
};

export const capitular = {
  viewBox: `0 0 ${CAJA.w} ${CAJA.h}`,
  tinta: TRAZOS.map(cinta),
  guias: TRAZOS.map((t) => dCurva(t.curvas)),
  anchos: TRAZOS.map((t) => t.ancho + 8),
  cola: dCurva(COLA),
  entrada: [TRAZOS[0].curvas[0], TRAZOS[0].curvas[1]] as Pt,
  salida: [COLA[COLA.length - 2], COLA[COLA.length - 1]] as Pt,
};
