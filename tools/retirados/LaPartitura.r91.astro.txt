      // archivos: el contorno del agente que lo abrió; un destello cuando la página toma algo de él
      archivos.forEach((e, f) => {
        let o = 0, ag = 0, d = 0;
        for (const a of ABRIR) if (a.f === f && tb >= a.at) { o = sm(tb, a.at, a.at + 0.2); ag = a.ag; }
        for (const x of DESTELLO) if (x.f === f) d = Math.max(d, blip(tb, x.at, 0.8));
        pon(e, '--o', f3(o));
        pon(e, '--c', AG[ag]);
        pon(e, '--d', f3(Math.max(d, o * 0.45)));
      });
      principios.forEach((e, j) => pon(e, '--p', f3(prog(t, PRIN(j), PRIN(j) + PRIN_D))));
      insignias.forEach((e, j) => {
        pon(e, '--f', f3(EZ.sale(prog(t, FORMA(j), FORMA(j) + 0.35))));
        pon(e, '--k', f3(sm(t, COLOR(j), COLOR(j) + 0.3)));
      });
      for (const [n, at] of Object.entries(PIEZA)) if (n !== 'titulo') pon(piezas[n], '--p', f3(EZ.sale(prog(tb, at, at + (n === 'campo' ? 1.2 : 0.55)))));
      palabras.forEach((e, q) => pon(e, '--p', f3(EZ.sale(prog(tb, PIEZA.titulo + q * 0.1, PIEZA.titulo + q * 0.1 + 0.5)))));
      pon(publicar, '--p', f3(sm(tb, PUBLICA, PUBLICA + 0.2)));
    };

    // ── el reloj
    let t0 = 0, raf = 0, jugando = false, tPrev = -1, tQuieto = 0, quietoDesde = 0, buscado: number | null = null;
    const tick = (ahora: number) => {
      raf = 0;
      if (!jugando || buscado !== null) return;
      const t = (((ahora - t0) / 1000) % LOOP + LOOP) % LOOP;
      if (tPrev >= 0 && ((t - tPrev + LOOP) % LOOP) < 0.25) sonar(tPrev, t);
      tPrev = t;
      pintar(t);
      raf = requestAnimationFrame(tick);
    };
    const tAhora = () => (((performance.now() - t0) / 1000) % LOOP + LOOP) % LOOP;

    montar();
    pintar(0);
    const rehacer = () => { montar(); pintar(buscado ?? (jugando ? tAhora() : 0)); };
    mqV.addEventListener('change', rehacer);
    let anchoPrevio = cine.clientWidth;
    new ResizeObserver(() => { if (Math.abs(cine.clientWidth - anchoPrevio) > 1) { anchoPrevio = cine.clientWidth; rehacer(); } }).observe(cine);
    document.fonts?.ready.then(rehacer);

    // sonda para las pruebas: __partitura.ir(t) congela el fotograma t; __partitura.seguir() vuelve al bucle
    (window as any).__partitura = {
      LOOP, BEAT,
      ir: (t: number) => { buscado = t; pintar(t); },
      seguir: () => { buscado = null; t0 = performance.now(); tPrev = -1; if (jugando && !raf) raf = requestAnimationFrame(tick); },
      t: () => (buscado ?? tAhora()),
    };

    if (reducido) { sec.classList.add('quieta'); return; }
    new IntersectionObserver((e) => {
      const ver = e[0].isIntersecting;
      if (ver && !jugando) {
        jugando = true;
        // si ha estado fuera más de 2 s, vuelve a empezar (siempre desde el póster); si no, sigue donde iba
        t0 = performance.now() - (performance.now() - quietoDesde > 2000 ? 0 : tQuieto * 1000);
        tPrev = -1;
        if (!raf) raf = requestAnimationFrame(tick);
      } else if (!ver && jugando) {
        jugando = false;
        tQuieto = tAhora();
        quietoDesde = performance.now();
      }
    }, { rootMargin: '0px 0px -12% 0px' }).observe(cine);
  }
</script>
