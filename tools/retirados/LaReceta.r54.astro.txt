import { receta } from '../data/home';
const pad = (n: number) => String(n + 1).padStart(2, '0');
// cada ticket, un poco distinto: giro mínimo (regla de la casa: nada torcido que reste seriedad) y altura
const GIRO = [-1.4, 0.9, -0.6, 1.2, -0.8];
const BAJA = [2.5, 0, 4, 1, 3];
// dónde aterriza lo digital en cada ticket (alto desde arriba · cuánto asoma por la derecha): cada uno en su sitio, nunca en fila
const DIG = [[58, -16], [30, -26], [74, -12], [44, -22], [82, -6]];
// la tinta A MANO (Coral): trazos con pulso (no geométricos), en una caja de 100 × 20 que se estira a la línea que marcan
const TRAZO: Record<string, string> = {
  circulo: 'M10 5 C 34 -1, 82 0, 96 7 C 103 13, 86 19.5, 50 19 C 17 18.6, 0 15.5, 3 9.5 C 5.5 4.5, 15 2.2, 32 2.4',
  subraya: 'M1 18.5 C 22 16.8, 48 19.6, 74 17.6 S 96 17, 99 16.2',
  tacha: 'M1 11.5 C 24 9.4, 52 12.4, 76 10.2 S 95 10.6, 99 9.2',
};

<!--
  LA RECETA DE PAPEL (rondas 53–54). Ronda 54, Andoni: «que caiga en un color claro y los tickets tengan más personalidad y parezcan más humanos. En cada
  ticket superponer algo digital que ayude a enfatizar lo que pasa en cada uno. También es un proceso de izquierda a derecha: mostrarlo mejor. El árbol de
  fondo va fuera». Así:
  · PROCESO: arriba de cada ticket su paso (01 tu marca → 05 la cuenta) sobre un hilo de savia que avanza de izquierda a derecha; los tickets se imprimen
    UNO TRAS OTRO en ese orden (no todos a la vez).
  · HUMANO: papel térmico con el canto de abajo dentado (arrancado), casi recto, y encima TINTA A MANO en Coral que se dibuja después de imprimir: un
    círculo, un subrayado, un tachado, marcas de visto (quien lo ha leído y lo ha usado).
  · DIGITAL: cuando el ticket ya está impreso y marcado, aterriza encima una pieza de INTERFAZ, recta y nítida (el contraste con el papel es la idea: el
    oficio y la IA): las muestras de la paleta · un bloqueo del arnés · el agente trabajando · la decisión guardada · publicado.
  El contenido de cada ticket es la receta REAL entregada a Neety (su library).
-->
<section id="la-savvia" class="surface-light receta" aria-labelledby="receta-h" data-receta data-sube>
  <div class="wrap section-y">
    <h2 id="receta-h" class="text-h1 max-w-[14ch]" data-entra>{receta.titulo}</h2>

    <div class="proceso mt-16 lg:mt-24" data-proceso>
      <span class="hilo-p" aria-hidden="true"><i data-hilo-p></i></span>
      {
        receta.tiras.map((t, i) => (
          <article class="tira" style={`--g:${GIRO[i]}deg;--b:${BAJA[i]}rem;--i:${i}`} data-tira aria-label={`${pad(i)} · ${t.paso}: ${t.t}`}>
            <p class="paso detail" aria-hidden="true"><b>{pad(i)}</b> {t.paso}</p>
            <div class="giro">
              <div class="papel">
                <p class="cab">{t.t}</p>
                <p class="raya" aria-hidden="true">{t.raya}</p>
                <div class="cuerpo">
                  {t.lineas.map((l, k) => {
                    const m = t.marcas.find(([n]) => n === k)?.[1];
                    return (
                      <span class="ln">
                        <span class="tx">
                          {l || ' '}
                          {m && m !== 'visto' && (
                            <svg class={`tinta ${m}`} viewBox="0 0 100 20" preserveAspectRatio="none" aria-hidden="true"><path d={TRAZO[m]} pathLength="1" /></svg>
                          )}
                          {m === 'visto' && (
                            <svg class="tinta visto" viewBox="0 0 16 16" aria-hidden="true"><path d="M2.2 8.6 C 3.6 9.6, 5 11.2, 6.1 13 C 8 8.6, 10.6 5, 14 2.4" pathLength="1" /></svg>
                          )}
                        </span>
                      </span>
                    );
                  })}
                </div>
              </div>
            </div>
            <div class={`dig dig-${t.digital.tipo}`} style={`--dy:${DIG[i][0]}%;--dx:${DIG[i][1]}%`} aria-hidden="true">
              {t.digital.tipo === 'muestras' && (
                <>
                  <span class="sws">{t.digital.colores!.map((c, k) => <i style={`--c:${c};--k:${k}`} />)}</span>
                  <b>{t.digital.txt}</b>
                </>
              )}
              {t.digital.tipo === 'bloqueo' && (
                <>
                  <span class="fila-d"><i class="x">✕</i>{t.digital.txt}</span>
                  <span class="fin">→ {t.digital.fin}</span>
                </>
              )}
              {t.digital.tipo === 'agente' && (
                <>
                  <span class="fila-d trabaja"><i class="giro-d" />{t.digital.txt}</span>
                  <span class="fila-d listo">{t.digital.fin}</span>
                </>
              )}
              {t.digital.tipo === 'commit' && (
                <span class="fila-d"><i class="punto" />{t.digital.txt}<b class="mas">{t.digital.fin}</b></span>
              )}
              {t.digital.tipo === 'toast' && <span class="fila-d">{t.digital.txt} <b>{t.digital.fin}</b></span>}
            </div>
          </article>
        ))
      }
    </div>
    <p class="nota detail mt-16 lg:mt-24">
      <span>{receta.nota}</span>
      <a class="link" href={receta.enlace.href} target="_blank" rel="noopener">{receta.enlace.label} →</a>
    </p>
  </div>
</section>

<script>
  const sec = document.querySelector<HTMLElement>('[data-receta]');
  if (sec) {
    const reducido = matchMedia('(prefers-reduced-motion: reduce)').matches;
    const tiras = [...sec.querySelectorAll<HTMLElement>('[data-tira]')];
    const hilo = sec.querySelector<HTMLElement>('[data-hilo-p]')!;
    const N = tiras.length;
    // EL PROCESO: al asomar la fila, cada ticket se imprime en su turno (0,9 s entre uno y otro), el hilo avanza hasta él, la tinta se dibuja
    // al acabar de imprimir y la pieza digital aterriza después
    const correr = () => {
      tiras.forEach((t, k) => {
        const a = k * 900;
        setTimeout(() => { t.classList.add('impresa'); hilo.style.setProperty('--p', String(N > 1 ? k / (N - 1) : 1)); }, a);
        setTimeout(() => t.classList.add('marcada'), a + 1500);
        setTimeout(() => t.classList.add('digital'), a + 2000);
      });
    };
    if (reducido) { tiras.forEach((t) => t.classList.add('impresa', 'marcada', 'digital')); hilo.style.setProperty('--p', '1'); }
    else {
      const io = new IntersectionObserver((es) => { if (es[0].isIntersecting) { io.disconnect(); correr(); } }, { threshold: 0.25 });
      io.observe(sec.querySelector('[data-proceso]')!);
    }
    (window as any).__receta = () => tiras.map((t) => [t.classList.contains('impresa'), t.classList.contains('marcada'), t.classList.contains('digital')]);
  }
</script>
