    // un tramo de A a B: sale y llega en vertical (una raíz que baja) y a mitad hace un RIZO, hacia el lado al que va
    const tramo = (A: P, B: P, rizo: boolean, lado: number) => {
      const out: P[] = [], dy = B.y - A.y, m = dy * 0.5, n = Math.max(12, Math.ceil(Math.hypot(B.x - A.x, dy) / 5));

        const t = i / n, u = 1 - t;
        let x = u * u * u * A.x + 3 * u * u * t * A.x + 3 * u * t * t * B.x + t * t * t * B.x;

          const th = ((t - 0.46) / 0.08) * Math.PI * 2;
          x += lado * r * Math.sin(th); y -= r * 0.95 * (1 - Math.cos(th)) * 0.5;
        }
        out.push({ x, y });
      }
      return out;

      gota.style.opacity = l > 8 && l < total - 8 ? '1' : '0';
      ultL = l;
      llegadas.forEach((k, n) => {
        if (i < k || ramas[n].sana) return;
        const p = pts[k];
        if (!reducido) salpicar(sec, p.x, p.y, { n: 7, fuerza: 0.55 });
        sanar(ramas[n], [sr.left + p.x, sr.top + p.y]);
      });
      if (l >= total - 2 && !hecho) { hecho = true; }
      titular();
    };
    const titular = () => {}; // ronda 76: el titular de Proyectos ya no se genera con ondas (ver global.css)
    let raf = 0;
    const pedir = () => { if (!raf) raf = requestAnimationFrame(pintar); };
    addEventListener('scroll', pedir, { passive: true });
    new ResizeObserver(() => medir()).observe(sec);
    ramas.forEach((r) => r.img.addEventListener('load', () => medir(), { once: true }));
    document.fonts.ready.then(medir);
    medir();
    const sonda = () => ({ sanas: ramas.map((r) => r.sana), vivas: ramas.map((r) => !!r.tarjeta?.classList.contains('viva')), total: Math.round(total), llegadas: llegadas.map((k) => Math.round(acum[k])), dibujado: Math.round(total + 20 - parseFloat(guia.style.strokeDashoffset || '0')), hecho });
    ((window as any).__dolores ??= {})[sec.id] = sonda;
  });
</script>
