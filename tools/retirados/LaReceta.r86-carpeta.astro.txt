// LA PARTITURA · RONDA 85 (2.ª). Andoni: «lo de Neety me gusta mucho el estilo, pero podemos hacer que sea una PARTITURA y que salga de algo relacionado con la
// música; un sobre no está muy relacionado. Y una vez hecha la animación, se queda como está y no vuelve a hacerla a la inversa cuando subes». Antes: la entrada
// en su sobre lacrado (tools/retirados/LaReceta.r85-sobre.astro.txt) y, antes, la impresora (…r84-impresora…).
// CONCEPTO: una CARPETA DE PARTITURAS de piel berenjena, atada con un cordón y con el isotipo grabado en naranja. Con el scroll: el cordón se suelta, la tapa se
// ABRE como un libro y dentro está LA PARTITURA de la web de Neety: portada clásica (savvia presenta · Neety · La partitura de su web) y los cinco MOVIMIENTOS, cada
// uno con su pentagrama y sus notas. Se recorren uno a uno (suenan y, al lado, su pieza digital: lo que hace esa parte en su web); al final se FIRMA (la v de
// savvia, a pluma) y se cierra con la doble barra final. Lo hecho se queda hecho: el avance nunca retrocede (al subir, la partitura sigue abierta).
import { receta } from '../data/home';
import isoRaw from '../../public/savvia-isotipo.svg?raw';
const ISO = isoRaw.match(/ d="([^"]+)"/)![1];
const E = receta.entrada;
const f = (n: number) => n.toFixed(2);
// la GUILLOCHE de las bandas (la del papel moneda)
const tira = (() => {
  const W = 48, H = 12, ps: string[] = [];
  for (const [fase, amp] of [[0, 4.6], [Math.PI, 4.6], [Math.PI / 2, 2.4], [(3 * Math.PI) / 2, 2.4]]) {
    let d = ''; for (let x = 0; x <= W; x += 1) d += (x ? 'L' : 'M') + f(x) + ' ' + f(H / 2 + amp * Math.sin((x / W) * Math.PI * 4 + fase));
    ps.push(`<path d='${d}' fill='none' stroke='%23431b44' stroke-width='.45' opacity='.75'/>`);
  }
  return `url("data:image/svg+xml;utf8,<svg xmlns='http://www.w3.org/2000/svg' width='${W}' height='${H}' viewBox='0 0 ${W} ${H}'>${ps.join('')}</svg>")`;
})();
// el INCIPIT de cada movimiento: un pentagrama corto con sus primeras notas (con semilla: siempre las mismas)
const incipit = (k: number) => {
  let sem = 31 + k * 97; const az = () => { sem = (sem * 16807) % 2147483647; return sem / 2147483647; };
  const W = 220, g = 5, y0 = 6, out: string[] = [];
  for (let l = 0; l < 5; l++) out.push(`<path class="l" d="M0 ${y0 + l * g}H${W}"/>`);
  out.push(`<path class="l" d="M${W} ${y0}V${y0 + 4 * g}"/>`);
  const n = 7 + Math.floor(az() * 4);
  for (let i = 0; i < n; i++) {
    const x = 18 + (i * (W - 30)) / n, p = Math.floor(az() * 9), y = y0 + 4 * g - (p * g) / 2, arriba = p < 4;
    out.push(`<ellipse class="c" cx="${f(x)}" cy="${f(y)}" rx="2.9" ry="2.1" transform="rotate(-22 ${f(x)} ${f(y)})"/><path class="p" d="M${f(x + (arriba ? 2.6 : -2.6))} ${f(y)}V${f(y + (arriba ? -15 : 15))}"/>`);
  }
  return `<svg viewBox="-2 0 ${W + 4} ${y0 * 2 + 4 * g}" preserveAspectRatio="none" aria-hidden="true">${out.join('')}</svg>`;
};
const romano = (m: string) => (m.includes('.') ? m.split('.')[0] : '');
const tempo = (m: string) => (m.includes('.') ? m.split('.')[1].trim() : m);
const cap = (t: string) => t.charAt(0).toUpperCase() + t.slice(1);

<script>
  import { tocar, pizzicato, papel } from '../scripts/musica';
  const sec = document.querySelector<HTMLElement>('[data-receta]');
  if (sec) {
    const reducido = matchMedia('(prefers-reduced-motion: reduce)').matches;
    const escena = sec.querySelector<HTMLElement>('[data-sala-entrada]')!, mesa = sec.querySelector<HTMLElement>('[data-mesa]')!;
    const movs = [...sec.querySelectorAll<HTMLElement>('[data-mov]')], digs = [...sec.querySelectorAll<HTMLElement>('[data-dig]')];
    const N = movs.length;
    const MELODIA = [62, 66, 69, 71, 74]; // una nota por movimiento (sube y resuelve)
    const final = () => { mesa.classList.add('abierta', 'fuera-carpeta', 'firmada'); movs.forEach((m) => m.classList.add('visto')); digs[N - 1].classList.add('on'); mesa.style.setProperty('--abrir', '1'); mesa.style.setProperty('--seva', '1'); };
    if (reducido) { mesa.classList.add('quieta'); final(); }
    else {
      // LO HECHO SE QUEDA HECHO: todo cuelga de pMax, el avance máximo alcanzado; al subir no se deshace nada
      let pMax = 0, estado = { suelto: false, abierta: false, activo: -1, firmada: false }, raf = 0;
      const clamp = (x: number) => Math.min(1, Math.max(0, x));
      const pintar = () => {
        raf = 0;
        const r = escena.getBoundingClientRect(), recorrido = r.height - innerHeight;
        pMax = Math.max(pMax, clamp(-r.top / Math.max(1, recorrido)));
        const p = pMax;
        // 1 · se suelta el cordón · 2 · se abre la tapa · 3 · la carpeta se retira y queda la partitura
        const suelta = clamp((p - 0.02) / 0.06), abrir = clamp((p - 0.07) / 0.13), seva = clamp((p - 0.2) / 0.1);
        mesa.style.setProperty('--suelta', suelta.toFixed(3)); mesa.style.setProperty('--abrir', abrir.toFixed(3)); mesa.style.setProperty('--seva', seva.toFixed(3));
        if (suelta > 0.1 && !estado.suelto) { estado.suelto = true; pizzicato(57, { vol: 0.08 }); }
        if (abrir > 0.05 && !estado.abierta) { estado.abierta = true; mesa.classList.add('abierta'); papel({ vol: 0.04, dur: 0.3 }); }
        mesa.classList.toggle('fuera-carpeta', seva > 0.99);
        // 4 · los movimientos, uno a uno (y su pieza al lado); el último se queda encendido
        const k = p < 0.32 ? -1 : Math.min(N - 1, Math.floor((p - 0.32) / (0.52 / N)));
        if (k > estado.activo) {
          estado.activo = k;
          movs.forEach((m, i) => { m.classList.toggle('on', i === k); m.classList.toggle('visto', i < k); });
          digs.forEach((d, i) => d.classList.toggle('on', i === k));
          tocar(MELODIA[k], { vol: 0.08, dur: 2 });
        }
        // 5 · la firma y la doble barra final
        if (p > 0.88 && !estado.firmada) { estado.firmada = true; mesa.classList.add('firmada'); [50, 62, 66, 69].forEach((m, i) => tocar(m, { vol: i ? 0.05 : 0.08, dur: 2.8, cuando: i * 0.06 })); }
      };
      const pedir = () => { if (!raf) raf = requestAnimationFrame(pintar); };
      let cerca = false;
      new IntersectionObserver((e) => { cerca = e[0].isIntersecting; if (cerca) pedir(); }, { rootMargin: '20% 0px' }).observe(escena);
      addEventListener('scroll', () => cerca && pedir(), { passive: true });
      addEventListener('resize', pedir);
      // si la página se abre ya por debajo (recarga a media página), la partitura aparece hecha
      if (escena.getBoundingClientRect().bottom < 0) { pMax = 1; final(); }
      pintar();
    }
    (window as any).__partitura = () => ({ clases: mesa.className, activo: movs.findIndex((m) => m.classList.contains('on')), p: mesa.style.getPropertyValue('--abrir') });
  }
</script>
