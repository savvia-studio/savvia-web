<script>
  import { crearEstela } from '../scripts/estela';
  import { NOCHE } from '../scripts/estacion';
  import { golpe } from '../scripts/musica';
  const hero = document.querySelector<HTMLElement>('[data-hero]');
  if (hero) {
    const img = hero.querySelector<HTMLImageElement>('[data-lamina]')!;
    const reducido = matchMedia('(prefers-reduced-motion: reduce)').matches;
    // RONDA 15: la gota la trae la carga (Cargador.astro): resbala por las letras de «savvia», cae aquí y avisa con `savvia:gota`.
    // Al tocar el suelo: la savia corre por la línea, el árbol se DIBUJA y se PINTA con canto rasgado; luego sangra el halo.
    // RONDA 66 · cada estación tiene su lámina: el motor ya no tiñe la hoja (intensidad 0), el color viene pintado
    let cv = hero.querySelector<HTMLCanvasElement>('[data-lienzo]')!;
    // RONDA 67 · «en primavera me gustaba el tono rosáceo, estilo cerezo»: en primavera el motor vuelve a teñir la hoja (su color de la tabla, rosa)
    // RONDA 83 · el director: sin tinte ni raleo de estación (el raleo del invierno se comía la piel y los puños: lo «saturado»)
    // se DIBUJA desde la punta de la batuta (el acorde de la carga la «da»): frente radial, sin gota
    let t = crearEstela(cv, img, { origen: [0.632, 0.02], dir: -1, radial: true, sinGota: true, tramado: true, tramadoBpm: 46, tramadoFuerza: 0.55, soplo: true, gota: true, manual: true, hd: true, intensidad: 0 });
    const w = window as any;
    // RONDA 30: al caer la gota, primero ENTRA EL HAZ desde arriba (luz-entra); la luz sobre el árbol la pone el shader al terminar de pintarse
    const suelta = () => { t?.soltar(); hero.classList.add('luz-entra'); setTimeout(() => hero.classList.add('crecido'), reducido ? 0 : 4200); };
    if (w.__gotaSuelta) suelta(); else addEventListener('savvia:gota', suelta, { once: true });
    if (!t) hero.classList.add('sin-gl');
    const caja = hero.querySelector<HTMLElement>('[data-arbol]')!;
    // la hora del estudio (Donostia), al minuto
    const hora = hero.querySelector<HTMLElement>('[data-hora]')!;
    const fmt = new Intl.DateTimeFormat('es-ES', { hour: '2-digit', minute: '2-digit', timeZone: 'Europe/Madrid' });
    // ronda 66: «solo la hora, sin Donostia»
    const tic = () => { hora.textContent = fmt.format(new Date()); setTimeout(tic, 60000 - (Date.now() % 60000) + 50); };
    tic();
    if (t) t.fondo(...NOCHE[0]); // ronda 85: la web ya no tiene estación (el Cacao de siempre)
    let luzV = 0;
    // RONDA 86 · «los puntos detrás de la ilustración también se ven afectados por la música»: la retícula del recuadro es un lienzo; a cada pulso (92 BPM) sale de
    // la batuta un anillo que agranda los puntos a su paso, y cualquier nota que suena en la web (musica.ts › golpe) los enciende un momento
    const marco = hero.querySelector<HTMLElement>('.marco')!, pc = marco.querySelector<HTMLCanvasElement>('[data-puntos]')!, px = pc.getContext('2d')!;
    let heroVisto = true, rafP = 0;
    const puntos = (ahora: number) => {
      rafP = 0;
      if (!heroVisto) return;
      const r = marco.getBoundingClientRect(), dpr = Math.min(2, devicePixelRatio || 1);
      if (pc.width !== Math.round(r.width * dpr) || pc.height !== Math.round(r.height * dpr)) { pc.width = Math.round(r.width * dpr); pc.height = Math.round(r.height * dpr); }
      px.setTransform(dpr, 0, 0, dpr, 0, 0); px.clearRect(0, 0, r.width, r.height);
      const cr = caja.getBoundingClientRect(), bx = cr.left - r.left + cr.width * 0.632, by = cr.top - r.top + cr.height * 0.02; // la batuta, en el recuadro
      const fase = ((ahora / 1000) * (46 / 60)) % 1, radio = fase * Math.hypot(r.width, r.height) * 0.7, amp = (1 - fase) * (reducido ? 0 : 0.55), gl = reducido ? 0 : golpe(); // ronda 87: «las ondas, más suaves»
      px.fillStyle = '#eadde7';
      for (let y = 4, fila = 0; y < r.height; y += 4, fila++) {
        const baja = 1 - y / (r.height * 0.96); if (baja <= 0) break;
        for (let x = fila % 2 ? 4 : 0; x < r.width; x += 8) {
          const d = Math.hypot(x - bx, y - by), onda = Math.exp(-Math.pow((d - radio) / 70, 2)) * amp;
          const rr = 1.15 + onda * 1.1 + gl * 0.9 * Math.exp(-d / 420);
          px.globalAlpha = Math.min(0.6, (0.1 + onda * 0.22 + gl * 0.2) * baja);
          px.beginPath(); px.arc(x, y, rr, 0, 6.283); px.fill();
        }
      }
      px.globalAlpha = 1;
      rafP = window.setTimeout(() => requestAnimationFrame(puntos), 33) as unknown as number;
    };
    // LA MAGIA: cada mano (izquierda 0,21 · 0,12; derecha, la de la batuta, 0,74 · 0,15) suelta motas de luz que viajan en curva hasta la pantalla de sus atriles
    // (0,11 · 0,30 a la izquierda; 0,69 · 0,89 a la derecha, a 0,62 de alto); al llegar, el atril se enciende un poco
    const mg = hero.querySelector<HTMLCanvasElement>('[data-magia]')!, mc = mg.getContext('2d')!;
    const MANOS = [[0.21, 0.12], [0.74, 0.15]], ATRILES = [[0.11, 0.62, 0], [0.3, 0.635, 0], [0.69, 0.635, 1], [0.89, 0.62, 1]];
    type Mota = { a: number; t: number; v: number; r: number; lado: number; ruido: number };
    const motas: Mota[] = [], brillo = [0, 0, 0, 0];
    let ultM = 0;
    const magia = (ahora: number) => {
      if (!heroVisto || reducido) return;
      if (!hero.classList.contains('crecido')) { setTimeout(() => requestAnimationFrame(magia), 400); return; } // espera a que el director esté pintado
      const r = mg.getBoundingClientRect(), dpr = Math.min(2, devicePixelRatio || 1);
      if (mg.width !== Math.round(r.width * dpr)) { mg.width = Math.round(r.width * dpr); mg.height = Math.round(r.height * dpr); }
      mc.setTransform(dpr, 0, 0, dpr, 0, 0); mc.clearRect(0, 0, r.width, r.height);
      const dt = ultM ? Math.min(0.05, (ahora - ultM) / 1000) : 0; ultM = ahora;
      if (motas.length < 70 && Math.random() < 0.6) { const a = Math.floor(Math.random() * 4); motas.push({ a, t: 0, v: 0.22 + Math.random() * 0.18, r: 0.5 + Math.random() * 1.2, lado: (Math.random() - 0.5) * 0.08, ruido: Math.random() * 6.28 }); }
      mc.globalCompositeOperation = 'lighter';
      for (let i = motas.length - 1; i >= 0; i--) {
        const m = motas[i]; m.t += m.v * dt;
        if (m.t >= 1) { brillo[m.a] = Math.min(1, brillo[m.a] + 0.12); motas.splice(i, 1); continue; }
        const [ax, ay, mano] = ATRILES[m.a], [hx, hy] = MANOS[mano];
        // curva: sale hacia arriba y hacia fuera de la mano y cae sobre el atril
        const cx2 = (hx + ax) / 2 + (ax < hx ? -0.08 : 0.08) + m.lado, cy2 = Math.min(hy, ay) - 0.16;
        const u = m.t, x = ((1 - u) * (1 - u) * hx + 2 * (1 - u) * u * cx2 + u * u * ax) * r.width + Math.sin(ahora / 300 + m.ruido) * 2;
        const y = ((1 - u) * (1 - u) * hy + 2 * (1 - u) * u * cy2 + u * u * ay) * r.height + Math.cos(ahora / 360 + m.ruido) * 2;
        const al = Math.sin(Math.PI * u) * 0.75;
        const gr = mc.createRadialGradient(x, y, 0, x, y, m.r * 4);
        gr.addColorStop(0, `rgba(255,226,170,${al})`); gr.addColorStop(0.4, `rgba(255,150,90,${al * 0.35})`); gr.addColorStop(1, 'rgba(255,120,80,0)');
        mc.fillStyle = gr; mc.beginPath(); mc.arc(x, y, m.r * 4, 0, 6.283); mc.fill();
      }
      // los atriles, encendidos por la magia que llega
      ATRILES.forEach(([ax, ay], k) => { if (brillo[k] < 0.01) return; const x = ax * r.width, y = ay * r.height, R = r.width * 0.07; const gr = mc.createRadialGradient(x, y, 0, x, y, R); gr.addColorStop(0, `rgba(255,214,150,${0.35 * brillo[k]})`); gr.addColorStop(1, 'rgba(255,214,150,0)'); mc.fillStyle = gr; mc.beginPath(); mc.arc(x, y, R, 0, 6.283); mc.fill(); brillo[k] *= 0.96; });
      mc.globalCompositeOperation = 'source-over';
      requestAnimationFrame(magia);
    };
    new IntersectionObserver((e) => { if (e[0].isIntersecting) { ultM = 0; requestAnimationFrame(magia); } }).observe(mg);
    new IntersectionObserver((e) => { t?.visible(e[0].isIntersecting); heroVisto = e[0].isIntersecting; if (heroVisto && !rafP) requestAnimationFrame(puntos); }).observe(hero);
    // la luz sigue al ratón (con retraso); con movimiento reducido o en táctil se queda quieta
    // RONDA 29 · «tiene que parecer que el haz incide en el árbol, no al revés»: el brillo del árbol (shader) se coloca EXACTAMENTE bajo el haz
    // principal del hero: misma inclinación (18°: tan = 0,325, la del shader) y misma x a la altura de la copa, con el ratón incluido
    const alinear = (mx: number) => {
      if (!t) return;
      const hr = hero.getBoundingClientRect(), cr = caja.getBoundingClientRect();
      const base = hr.width * 0.57 + mx * 0.38 * innerWidth; // centro del haz «a» (left 40 % + ancho 34 %) en el hero
      const xTop = base - 0.325 * (cr.top - hr.top - hr.height * 0.5); // el haz gira sobre su centro (a media altura del hero)
      const d = (xTop - (cr.left - hr.left)) / cr.width; // en fracciones de la caja del árbol
      luzV = 0.95 * d - 0.62; t.luz(luzV);
    };
    alinear(0); addEventListener('resize', () => alinear(0));
    if (!reducido && matchMedia('(hover: hover)').matches) {
      let mx = 0, my = 0, tx = 0, ty = 0, raf = 0;
      const seguir = () => {
        raf = 0;
        mx += (tx - mx) * 0.05; my += (ty - my) * 0.05;
        hero.style.setProperty('--lx', mx.toFixed(4)); hero.style.setProperty('--ly', my.toFixed(4));
        alinear(mx);
        if (Math.abs(tx - mx) + Math.abs(ty - my) > 0.001) raf = requestAnimationFrame(seguir);
      };
      hero.addEventListener('pointermove', (e) => { const r = hero.getBoundingClientRect(); tx = (e.clientX - r.left) / r.width - 0.5; ty = (e.clientY - r.top) / r.height - 0.5; if (!raf) raf = requestAnimationFrame(seguir); }, { passive: true });
      hero.addEventListener('pointerleave', () => { tx = 0; ty = 0; if (!raf) raf = requestAnimationFrame(seguir); });
    }
  }
</script>
