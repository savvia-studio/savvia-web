// remates redondos y una BOLA al final del trazo fino). Cada instrumento se escribe ahora así: su trazo es una línea central (en 48 × 48) que `plumilla()`

type P = [number, number];
const muestrear = (d: string): P[] => {
  const t = d.match(/[MLCQ]|-?\d*\.?\d+/g)!;
  let i = 0, cmd = '', cur: P = [0, 0];
  const pts: P[] = [], n = () => parseFloat(t[i++]);
  const bez = (f: (u: number) => P, k: number) => { for (let s = 1; s <= k; s++) pts.push(f(s / k)); };
  while (i < t.length) {
    if (/[MLCQ]/.test(t[i])) cmd = t[i++];
    if (cmd === 'M') { cur = [n(), n()]; pts.push(cur); cmd = 'L'; }
    else if (cmd === 'L') { const a = cur, b: P = [n(), n()]; bez((u) => [a[0] + (b[0] - a[0]) * u, a[1] + (b[1] - a[1]) * u], 10); cur = b; }
    else if (cmd === 'Q') { const a = cur, c: P = [n(), n()], b: P = [n(), n()]; bez((u) => { const v = 1 - u; return [v * v * a[0] + 2 * v * u * c[0] + u * u * b[0], v * v * a[1] + 2 * v * u * c[1] + u * u * b[1]]; }, 20); cur = b; }
    else { const a = cur, c1: P = [n(), n()], c2: P = [n(), n()], b: P = [n(), n()]; bez((u) => { const v = 1 - u; return [v * v * v * a[0] + 3 * v * v * u * c1[0] + 3 * v * u * u * c2[0] + u * u * u * b[0], v * v * v * a[1] + 3 * v * v * u * c1[1] + 3 * v * u * u * c2[1] + u * u * u * b[1]]; }, 28); cur = b; }
  }
  return pts;
};
const f2 = (x: number) => +x.toFixed(2);
const circulo = (x: number, y: number, r: number) => `M${f2(x - r)} ${f2(y)}a${f2(r)} ${f2(r)} 0 1 0 ${f2(2 * r)} 0a${f2(r)} ${f2(r)} 0 1 0 ${f2(-2 * r)} 0`;
// la pluma: su canto va inclinado ~29° (0,5 rad); el trazo es grueso cuando cruza ese canto (verticales, la diagonal que sube) y fino cuando lo sigue
const plumilla = (d: string, { max = 4.2, min = 0.42, nib = 0.5, puntas = 0.1, k: escala = 1 } = {}) => {
  const p = muestrear(d), L: P[] = [], R: P[] = [], acu = [0];
  for (let k = 1; k < p.length; k++) acu.push(acu[k - 1] + Math.hypot(p[k][0] - p[k - 1][0], p[k][1] - p[k - 1][1]));
  const tot = acu[acu.length - 1] || 1;
  // el ancho de cada punto y, suavizado (tres pasadas), para que los quiebros no dejen escalones: la pluma gira, no salta
  const dir = p.map((_, k) => { const a = p[Math.max(0, k - 1)], b = p[Math.min(p.length - 1, k + 1)]; return [b[0] - a[0], b[1] - a[1]] as P; });
  let W = p.map((_, k) => {
    const [dx, dy] = dir[k], u = acu[k] / tot, af = Math.min(1, u / puntas, (1 - u) / puntas);
    // contraste fuerte, como el isotipo: el seno se eleva (los trazos «a medias» adelgazan) y las puntas se afilan hasta un hilo redondo
    return escala * (min + (max - min) * Math.abs(Math.sin(Math.atan2(dy, dx) - nib)) ** 1.7) * (0.35 + 0.65 * Math.sin((Math.min(1, af) * Math.PI) / 2));
  });
  for (let v = 0; v < 3; v++) W = W.map((w, k) => (W[Math.max(0, k - 2)] + W[Math.max(0, k - 1)] + w + W[Math.min(W.length - 1, k + 1)] + W[Math.min(W.length - 1, k + 2)]) / 5);
  const w0 = W[0], w1 = W[W.length - 1];
  p.forEach((q, k) => {
    const [dx, dy] = dir[k], l = Math.hypot(dx, dy) || 1, w = W[k];
    const nx = (-dy / l) * (w / 2), ny = (dx / l) * (w / 2);
    L.push([q[0] + nx, q[1] + ny]); R.push([q[0] - nx, q[1] - ny]);
  });
  const lado = (qs: P[]) => qs.map((q) => `${f2(q[0])} ${f2(q[1])}`).join('L');
  return `M${lado(L)}L${lado(R.reverse())}Z${circulo(p[0][0], p[0][1], w0 / 2)}${circulo(p[p.length - 1][0], p[p.length - 1][1], w1 / 2)}`;
};

<script>
  import { tocar, pizzicato, diapason } from '../scripts/musica';
  // RONDA 90 · el subtítulo: al entrar en pantalla se tacha «necesites» y salen las opciones (una vez); los instrumentos se dibujan trazo a trazo al verse
  const once = (els: Element[], cls: string, margen = '0px 0px -15% 0px') => {
    const o = new IntersectionObserver((es) => es.forEach((e) => { if (e.isIntersecting) { e.target.classList.add(cls); o.unobserve(e.target); } }), { rootMargin: margen });
    els.forEach((el) => o.observe(el));
  };
  // ronda 92 · la frase: al verse, se tacha «necesites» y salen del talonario las demás entradas (una vez)
  once([...document.querySelectorAll('[data-sub]')], 'ya', '0px 0px -20% 0px');
  once([...document.querySelectorAll('[data-servicios] .instrumento')], 'dibuja', '0px');
  const tabla = document.querySelector<HTMLElement>('[data-tabla]');
  if (tabla) {
    const banda = tabla.querySelector<HTMLElement>('[data-banda]')!, cabezas = [...tabla.querySelectorAll<HTMLElement>('.cabeza [data-c]')];
    // la banda de cristal se coloca sobre la columna activa (toda la altura de la tabla)
    const colocar = () => {
      const k = Number(tabla.dataset.col || 0), c = cabezas[k], t = tabla.getBoundingClientRect(), r = c.getBoundingClientRect();
      banda.style.transform = `translateX(${(r.left - t.left).toFixed(1)}px)`; banda.style.width = `${r.width.toFixed(1)}px`;
    };
    // RONDA 76 · Eric: «la interacción del cambio, no sé si está muy fina»: la columna cambia por INTENCIÓN (el puntero se queda 90 ms en ella), no al
    // rozarla al cruzar en diagonal; y al salir de la tabla vuelve a la que se miraba (sin saltos)
    let intento = 0;
    tabla.addEventListener('pointerover', (e) => {
      const c = (e.target as HTMLElement).closest<HTMLElement>('[data-c]');

    });

    vigilar(); addEventListener('resize', vigilar);

    const io = new IntersectionObserver((es) => es.forEach((e) => { if (e.isIntersecting) { e.target.classList.add('llega'); io.unobserve(e.target); } }), { rootMargin: '0px 0px -12% 0px' });
    tabla.querySelectorAll('[data-fila]').forEach((f) => io.observe(f));
