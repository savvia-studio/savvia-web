<script>
  import { tocar, sonidoEncendido, PENTA } from '../scripts/musica';
  import { crearPapel } from '../scripts/papel';
  const pie = document.querySelector<HTMLElement>('[data-pie]');
  if (pie) {
    const reducido = matchMedia('(prefers-reduced-motion: reduce)').matches;
    const NS = 'http://www.w3.org/2000/svg';
    const final = pie.querySelector<HTMLElement>('[data-final]')!, pauta = pie.querySelector<SVGSVGElement>('[data-pauta]')!;
    const gL = pie.querySelector<SVGGElement>('[data-lineas]')!, barra = pie.querySelector<SVGGElement>('[data-barra]')!;
    const contacta = pie.querySelector<HTMLAnchorElement>('[data-contacta]')!, txt = contacta.querySelector<HTMLElement>('[data-contacta-txt]')!;
    const lineas: SVGPathElement[] = [];
    for (let i = 0; i < 5; i++) { const l = document.createElementNS(NS, 'path'); l.setAttribute('pathLength', '1'); gL.appendChild(l); lineas.push(l); }
    // GEOMETRÍA: la palabra a todo el ancho del contenido; el pentagrama desde su línea base hasta la altura de sus mayúsculas (la «C»)
    let W = 0, x0 = 0, x1 = 0, yb = 0, g = 20, fs = 100, cx = 0, xsLetras: number[] = [];
    const medir = () => {
      const cw = final.clientWidth - 2 * parseFloat(getComputedStyle(final).paddingLeft);
      const c = document.createElement('canvas').getContext('2d')!;
      c.font = `100px ${getComputedStyle(contacta).fontFamily}`;
      const tw = c.measureText('Contáctanos').width || 1; // (las letras van en spans: se mide la palabra entera)
      fs = (100 * cw * 0.9) / tw; contacta.style.fontSize = `${fs}px`;
      const real = txt.getBoundingClientRect().width; if (real > 0) { fs *= (cw * 0.9) / real; contacta.style.fontSize = `${fs}px`; }
      const fr = final.getBoundingClientRect(), tr = txt.getBoundingClientRect();
      W = fr.width; pauta.setAttribute('viewBox', `0 0 ${W} ${fr.height}`);
      const pad = parseFloat(getComputedStyle(final).paddingLeft);
      x0 = pad; x1 = W - pad;
      // la línea base de Family ≈ el 78 % del alto de su caja de línea (line-height 1) · la altura de mayúscula ≈ 0,68 em
      yb = tr.top - fr.top + fs * 0.8; g = (fs * 0.68) / 4;
      cx = tr.left - fr.left + tr.width * 0.5;
      pintar(performance.now());
      // RONDA 95 · el director de perfil, a la izquierda y mirando a la palabra: lo más alto que quepa entre el subtítulo y la línea base (la palabra le tapa
      // de las rodillas abajo); en móvil, más pequeño
      const sub = pie.querySelector<HTMLElement>('.contenido .text-sub')!.getBoundingClientRect();
      const libre = yb - (sub.bottom - fr.top) - 16;
      const dh = Math.max(140, Math.min(libre / 0.78, fr.width < 600 ? fr.width * 0.7 : fs * 1.6)), dw = (dh * 1024) / 1536; // en móvil la palabra es pequeña: él puede crecer
      dirEl.style.width = `${dw.toFixed(0)}px`; dirEl.style.height = `${dh.toFixed(0)}px`;
      dirEl.style.left = `${(x0 - dw * 0.18).toFixed(0)}px`; dirEl.style.top = `${(yb - dh * 0.78).toFixed(0)}px`;
      contacta.style.setProperty('--fs', `${fs.toFixed(1)}px`);
      // la doble barra final
      const alto = `M0 ${(yb - 4 * g).toFixed(1)}V${yb.toFixed(1)}`;
      barra.querySelector('.fina')!.setAttribute('d', alto); barra.querySelector('.gruesa')!.setAttribute('d', alto);
      barra.setAttribute('transform', `translate(${x1.toFixed(1)} 0)`);
      const letras = [...txt.querySelectorAll<HTMLElement>('.l')];
      xsLetras = letras.map((l) => { const r = l.getBoundingClientRect(); return r.left - fr.left + r.width / 2; });
    };
    // la cuerda que vibra (como en la carga): onda estacionaria que se apaga
    const vib = lineas.map(() => ({ a: 0, t: 0 }));
    const pintar = (ahora: number) => {
      lineas.forEach((l, i) => {
        const y = yb - i * g, v = vib[i], t = (ahora - v.t) / 1000, A = v.a * Math.exp(-t * 2.6);
        if (A < 0.05) { l.setAttribute('d', `M${x0.toFixed(1)} ${y.toFixed(1)}H${x1.toFixed(1)}`); return; }
        let d = ''; const n = 64, L = x1 - x0;
        for (let k = 0; k <= n; k++) { const x = x0 + (L * k) / n; d += (k ? 'L' : 'M') + x.toFixed(1) + ' ' + (y + A * Math.sin((Math.PI * k) / n) * Math.sin(k * 0.5 - t * 34) * Math.cos(t * 19)).toFixed(2); }
        l.setAttribute('d', d);
      });
    };
    let raf = 0;
    const bucle = (ahora: number) => { pintar(ahora); raf = vib.some((v) => ahora - v.t < 2200) ? requestAnimationFrame(bucle) : 0; };
    const pulsar = () => { if (reducido) return; const ahora = performance.now(); vib.forEach((v, i) => { v.a = g * (0.28 - i * 0.02); v.t = ahora + i * 40; }); if (!raf) raf = requestAnimationFrame(bucle); };
    // el acorde final (ronda 85: uno solo, fa mayor: el de la carga; la web ya no tiene estaciones)
    const ACORDE = [53, 65, 69, 72];
    let sonado = 0;
    const abrir = () => {
      if (pie.classList.contains('cta-abierto')) return;
      pie.classList.add('cta-abierto'); pulsar();
      if (!reducido && sonidoEncendido() && performance.now() - sonado > 2500) {
        sonado = performance.now();
        ACORDE.forEach((m, i) => tocar(m, { vol: i ? 0.07 : 0.1, dur: 3.2, cuando: i * 0.05 }));
      }
    };
    const cerrar = () => pie.classList.remove('cta-abierto');
    // los enlaces del pie, como las cuerdas de un arpa (ronda 85)
    pie.querySelectorAll<HTMLElement>('.enlace-pie').forEach((a, i) => a.addEventListener('pointerenter', () => tocar(PENTA[(i * 2) % PENTA.length] + 12, { vol: 0.04, dur: 1.3 })));
    contacta.addEventListener('pointerenter', abrir);
    contacta.addEventListener('pointerleave', cerrar);
    contacta.addEventListener('focus', abrir);
    contacta.addEventListener('blur', cerrar);
    // RONDA 85 · «el Contáctanos del final: que sean NOTAS que se vayan tocando y se quede finalmente el Contáctanos». Al llegar se traza el pentagrama; luego,
    // letra a letra, una nota aparece sobre el pentagrama encima de su letra (una melodía que sube y baja), SUENA, se deshace y deja su letra. Al final se
    // cierra la doble barra y suena el acorde. Sin movimiento: la palabra, directamente.
    const MELODIA = [1, 3, 5, 4, 6, 8, 7, 5, 6, 4, 2]; // posición en el pentagrama (medios espacios desde la primera línea)
    const NATURAL = [64, 65, 67, 69, 71, 72, 74, 76, 77];
    const letras = [...txt.querySelectorAll<HTMLElement>('.l')];
    const llegar = () => {
      pie.classList.add('llega');
      if (reducido) { letras.forEach((l) => l.classList.add('on')); pie.classList.add('entera'); return; }
      // RONDA 85 (2.ª) · «que las propias letras sean las notas y sean lo que se queda»: cada LETRA entra a la altura de su nota en el pentagrama (sube o baja
      // según la melodía), con su plica, y suena; cuando han sonado todas, bajan juntas a la línea (el acorde) y queda la palabra
      letras.forEach((l, i) => setTimeout(() => {
        const p = MELODIA[i];
        l.style.setProperty('--y', `${(-(p - 3) * g * 0.42).toFixed(1)}px`);
        l.classList.add('on', 'nota');
        tocar(NATURAL[p], { vol: 0.08, dur: 1.6 });
        vib[Math.min(4, Math.round(p / 2))] = { a: g * 0.18, t: performance.now() }; if (!raf) raf = requestAnimationFrame(bucle);
        if (i === letras.length - 1) setTimeout(() => {
          pie.classList.add('entera'); letras.forEach((q) => { q.style.setProperty('--y', '0px'); q.classList.remove('nota'); });
          ACORDE.forEach((m, k) => tocar(m, { vol: k ? 0.06 : 0.09, dur: 3, cuando: k * 0.04 }));
          for (let k = 0; k < 5; k++) vib[k] = { a: g * 0.26, t: performance.now() + k * 40 };
          if (!raf) raf = requestAnimationFrame(bucle);
        }, 650);
      }, 1500 + i * 190));
    };
    // RONDA 94 · EL DIRECTOR del pie (se monta al acercarse; marca el compás solo a la vista)
    const dirEl = pie.querySelector<HTMLElement>('[data-director-pie]')!;
    let td: ReturnType<typeof crearPapel> | null = null;
    const montarDirector = () => {
      if (td) return;
      const img = new Image(); img.decoding = 'async'; img.src = '/media/pie/director-perfil.webp';
      const bg = (getComputedStyle(pie).backgroundColor.match(/\d+/g) || ['59', '35', '27']).slice(0, 3).map(Number) as [number, number, number];
      // el rig, medido en la lámina de perfil (1024 × 1536): los dos brazos salen del hombro hacia delante; la batuta, de la mano a la punta
      td = crearPapel(dirEl.querySelector('canvas')!, img, {
        origen: [0.928, 0.026], semilla: 19, densidad: 0.75, batuta: [0.928, 0.026, 0.775, 0.112], bpm: 46, amplitud: 0.9,
        brazos: [{ hombro: [0.5, 0.25], mano: [0.875, 0.2], ancho: 0.075 }, { hombro: [0.48, 0.215], mano: [0.78, 0.12], ancho: 0.075 }],
      });
      td.fondo(bg[0], bg[1], bg[2]);
      if (reducido) td.soltar();
    };
    new IntersectionObserver((e) => { if (e[0].isIntersecting) montarDirector(); td?.visible(e[0].isIntersecting); }, { rootMargin: '400px 0px' }).observe(dirEl);
    const pintarSala = () => td?.soltar(); // ronda 94: el director se compone desde la batuta al llegar
    new IntersectionObserver((e, io) => { if (e[0].isIntersecting) { montarDirector(); pintarSala(); llegar(); io.disconnect(); } }, { threshold: 0.3 }).observe(final);
    document.fonts.ready.then(medir); addEventListener('resize', medir); medir();
  }
</script>
