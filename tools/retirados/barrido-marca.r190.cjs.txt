// RONDA 190 · el BARRIDO DE MARCA de «Más proyectos» (Eric y Andoni, Telegram 14/10: a Andoni le gustan los degradados con los colores de cada
// proyecto, «no es todo plano e igual», y no los bordes difuminados; a Eric, el gradiente BARRIDO, «le da modernidad»). Punto en común: los tres
// colores de la web de cada cliente en manchas con canto, arrastradas con desenfoque de MOVIMIENTO (como tools/r189/barrido.cjs con el paisaje).
// Uso: node tools/r190/barrido-marca.cjs → public/media/anteriores/fondos/marca-<slug>.webp
const sharp = require('sharp');
const W = 900, H = 900, L = 240, PAD = L;
const hex = (h) => [1, 3, 5].map((i) => parseInt(h.slice(i, i + 2), 16));
const mezcla = (a, b, t) => a.map((v, i) => v + (b[i] - v) * t);
function azar(s) { return () => ((s = (s * 1664525 + 1013904223) >>> 0) / 4294967296); }
const P = [
  // [slug, colores (fondo · secundario · acento), ángulo del arrastre, semilla]
  ['kymatio', ['#0b1f17', '#123a2c', '#2fbf8a'], -24, 3],
  ['wisable', ['#0b0f2e', '#1b2150', '#e6f24a'], 20, 7],
  ['monad', ['#e4efe0', '#b8d8b1', '#86b98a'], -30, 11],
  ['transcend-network', ['#fff6ee', '#ff6b1a', '#3f7bff'], 24, 5],
  ['cubbo', ['#1b40d6', '#4963e2', '#788ae8'], -18, 13],
  ['remerge', ['#eceed9', '#f2ff71', '#ffffff'], 28, 17],
  ['thx', ['#3f86bd', '#7fb3dc', '#c9dff0'], -34, 19],
  // los destacados que rotan en la pieza de entrada
  ['h-kuanta', ['#15181c', '#f01438', '#3a3e44'], -36, 23],
  ['h-arkdev', ['#f8f4ee', '#e8542b', '#998a82'], -36, 23],
  ['h-playground', ['#21127a', '#5c2e6e', '#3d7a96'], -36, 23],
  ['h-neety', ['#EEE8F7', '#A7C5F9', '#FE8238'], -36, 23],
];
(async () => {
  for (const [slug, cs, ang, sem] of P) {
    if (process.argv[2] && process.argv[2] !== slug) continue;
    const [c1, c2, c3] = cs.map(hex), r = azar(sem), w = W + 2 * PAD, h = H + 2 * PAD;
    const img = new Float32Array(w * h * 3);
    for (let i = 0; i < w * h; i++) img.set(c1, i * 3);
    // FRANJAS a lo ancho del arrastre (como una cámara que barre): anchos distintos, canto blando o duro, que ondulan despacio a lo largo;
    // encima, hilos finos del acento. El desenfoque de movimiento las convierte en vetas.
    const A0 = (ang * Math.PI) / 180, px = -Math.sin(A0), py = Math.cos(A0), tx = Math.cos(A0), ty = Math.sin(A0);
    const franjas = []; let pos = -h;
    while (pos < h * 1.5) { const ancho = 30 + r() * 220; franjas.push({ a: pos, b: pos + ancho, c: [c1, c2, c3, mezcla(c1, c2, 0.5), c1, mezcla(c2, c3, 0.5)][Math.floor(r() * 6)], duro: r() < 0.45 }); pos += ancho; }
    const hilos = []; for (let k = 0; k < 14; k++) hilos.push({ p: (r() - 0.3) * h * 1.6, g: 1.5 + r() * 5, c: r() < 0.6 ? c3 : c2, o: 0.5 + r() * 0.5, f: r() * 6 });
    const ond = [r() * 6, r() * 6];
    for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) {
      const t = x * tx + y * ty, q = x * px + y * py + 44 * Math.sin(t * 0.0042 + ond[0]) + 30 * Math.sin(t * 0.011 + ond[1]);
      let col = c1;
      for (let k = 0; k < franjas.length; k++) {
        const f = franjas[k]; if (q < f.a || q >= f.b) continue;
        const sig = franjas[k + 1] ?? f, d = f.b - q, blando = f.duro ? 3 : 60;
        col = d < blando ? mezcla(sig.c, f.c, d / blando) : f.c; break;
      }
      const i = (y * w + x) * 3; img[i] = col[0]; img[i + 1] = col[1]; img[i + 2] = col[2];
      for (const hl of hilos) { const d = Math.abs(q - hl.p - 12 * Math.sin(t * 0.02 + hl.f)); if (d < hl.g) for (let c = 0; c < 3; c++) img[i + c] += (hl.c[c] - img[i + c]) * hl.o * (1 - d / hl.g); }
    }
    const A = (ang * Math.PI) / 180, ux = Math.cos(A), uy = Math.sin(A), out = Buffer.alloc(W * H * 3);
    for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) {
      let R = 0, G = 0, B = 0, n = 0;
      for (let k = -L / 2; k <= L / 2; k++) {
        const wg = 1 - Math.abs(k) / (L / 2 + 1), i = (Math.round(y + PAD + k * uy) * w + Math.round(x + PAD + k * ux)) * 3;
        R += img[i] * wg; G += img[i + 1] * wg; B += img[i + 2] * wg; n += wg;
      }
      const o = (y * W + x) * 3; out[o] = R / n; out[o + 1] = G / n; out[o + 2] = B / n;
    }
    await sharp(out, { raw: { width: W, height: H, channels: 3 } }).blur(1.4).webp({ quality: 80 }).toFile(`public/media/anteriores/fondos/marca-${slug}.webp`);
    console.log(slug);
  }
})();
