<script>
  import { rgbDe } from '../scripts/tinta';
  import { tocar } from '../scripts/musica';
  const raiz = document.querySelector<HTMLElement>('[data-equipo]');
  if (raiz) {
    const paseo = raiz.querySelector<HTMLElement>('[data-paseo]')!, a = raiz.querySelector<HTMLElement>('[data-a]')!, e = raiz.querySelector<HTMLElement>('[data-e]')!;
    const juntos = raiz.querySelector<HTMLElement>('[data-juntos]')!, quienes = [...raiz.querySelectorAll<HTMLButtonElement>('[data-quien]')];
    const reducido = matchMedia('(prefers-reduced-motion: reduce)').matches;
    // descripciones: una abierta cada vez
    const abrir = (i: number | null) => { quienes.forEach((b, k) => b.setAttribute('aria-expanded', String(k === i))); juntos.dataset.mira = i === null ? '' : String(i); };
    quienes.forEach((b, i) => {
      // RONDA 85 · al mirar a uno suena su motivo: Andoni, más grave; Eric, más agudo (como en la conversación de las preguntas)
      b.addEventListener('pointerenter', (ev) => { if (ev.pointerType !== 'mouse') return; abrir(i); const b0 = i ? 74 : 62; [0, 4, 7, 12].forEach((d, k) => tocar(b0 + d, { vol: 0.05, dur: 1.6, cuando: k * 0.09 })); });
      b.addEventListener('pointerleave', (ev) => ev.pointerType === 'mouse' && abrir(null));
      b.addEventListener('focus', () => abrir(i));
      b.addEventListener('blur', () => abrir(null));
      b.addEventListener('click', () => abrir(b.getAttribute('aria-expanded') === 'true' ? null : i));
    });
    // ── RONDA 86 · la estela, como la imagen de Andoni (tools/r85/estela-ref.jpg): una CINTA de muchas líneas finas que se RETUERCE (se estrecha y se abre, las
    //   líneas se juntan en el giro), en degradado de amarillo (junto a ellos) a naranja y rojo (lejos), que nace del borde y llega hasta sus pies, y ondula
    //   despacio como una onda sonora. Todo se apaga al juntarse.
    const cv = raiz.querySelector<HTMLCanvasElement>('[data-verde]')!, cx = cv.getContext('2d')!;
    raiz.querySelector('[data-verde-delante]')?.remove();
    let VW = 0, VH = 0, xaU = -1e6, xeU = 1e6, reloj = 0, avanceU = 0;
    const sembrar = () => {
      const r = cv.getBoundingClientRect(), dpr = Math.min(devicePixelRatio || 1, 2); VW = r.width; VH = r.height;
      cv.width = Math.round(VW * dpr); cv.height = Math.round(VH * dpr); cx.setTransform(dpr, 0, 0, dpr, 0, 0);
    };
    const LINEAS = 46;
    const cinta = (x0: number, x1: number, haciaDer: boolean, t: number, apaga: number) => {
      if (x1 - x0 < 20) return;
      const L = x1 - x0, g = cx.createLinearGradient(haciaDer ? x0 : x1, 0, haciaDer ? x1 : x0, 0); // de junto a ellos hacia el borde
      // de junto a ellos (amarillo) a lejos (rojo), y lo más lejano se desvanece
      g.addColorStop(0, `rgba(250,222,120,${0.5 * apaga})`); g.addColorStop(0.3, `rgba(250,158,70,${0.42 * apaga})`); g.addColorStop(0.7, `rgba(232,90,56,${0.3 * apaga})`); g.addColorStop(1, 'rgba(170,40,40,0)');
      cx.strokeStyle = g; cx.lineWidth = 0.8; cx.globalCompositeOperation = 'lighter'; // las líneas suman luz al juntarse (el giro brilla)
      const base = VH * 0.74, paso = 5;
      for (let i = 0; i < LINEAS; i++) {
        const u = i / (LINEAS - 1) - 0.5;
        cx.beginPath();
        for (let x = x0; x <= x1; x += paso) {
          const s = haciaDer ? (x1 - x) / L : (x - x0) / L; // 0 junto a ellos, 1 en el borde
          const c = base - Math.sin(s * 5.2 - t * 0.9) * VH * 0.12 * (0.3 + s) - (1 - s) * VH * 0.06;
          const giro = Math.cos(s * 7.5 - t * 0.6 + (haciaDer ? 0 : 1.3));
          const ancho = VH * (0.07 + 0.16 * s) * giro;
          const y = c + u * ancho + Math.sin(x * 0.02 + i * 0.4 + t) * 0.6;
          x === x0 ? cx.moveTo(x, y) : cx.lineTo(x, y);
        }
        cx.stroke();
      }
      cx.globalCompositeOperation = 'source-over';
    };
    const pintarVerde = (xa: number, xe: number, t = 0) => {
      xaU = xa; xeU = xe;
      cx.clearRect(0, 0, VW, VH);
      const apaga = Math.max(0, 1 - Math.pow(avanceU / 0.98, 3));
      if (apaga <= 0) return;
      cinta(0, Math.min(VW, xa - 24), false, t, apaga); // detrás de Andoni: del borde izquierdo a sus pies
      cinta(Math.max(0, xe + 24), VW, true, t, apaga); // detrás de Eric
    };
    // la brisa: solo mientras la sección se ve y ya ha brotado algo
    let viendo = false;
    const brisa = (ahora: number) => { reloj = 0; if (!viendo || reducido) return; pintarVerde(xaU, xeU, ahora / 1000); reloj = requestAnimationFrame(brisa); }; // las notas respiran
    new IntersectionObserver((en) => { viendo = en[0].isIntersecting; if (viendo && !reloj) reloj = requestAnimationFrame(brisa); }).observe(cv);
    sembrar();
    if (reducido) { raiz.classList.add('llegaron'); pintarVerde(1e6, -1e6); }
    if (!reducido) {
      let visible = false, pedido = false, ult = -1;
      const pintar = () => {
        pedido = false;
        const r = paseo.getBoundingClientRect(), vh = innerHeight, W = r.width, cw = a.offsetWidth;
        const lin = Math.min(1, Math.max(0, (vh * 0.96 - r.top) / (vh * 0.66)));
        if (Math.abs(lin - ult) < 0.001) return;
        ult = lin;
        // el último 20 % frena
        const p = lin < 0.8 ? lin : 0.8 + (1 - Math.pow(1 - (lin - 0.8) / 0.2, 2)) * 0.2;
        const fw = juntos.offsetWidth;
        const xa0 = -cw * 1.1, xa1 = W / 2 - fw * 0.27 - cw / 2, xe0 = W + cw * 0.1, xe1 = W / 2 + fw * 0.23 - cw / 2;
        const xa = xa0 + (xa1 - xa0) * p, xe = xe0 + (xe1 - xe0) * p;
        const paso = cw * 0.2, gira = lin > 0.93;
        a.style.transform = `translateX(${xa.toFixed(1)}px)`; e.style.transform = `translateX(${xe.toFixed(1)}px)`;
        a.style.backgroundPositionX = `${(gira ? 4 : Math.floor((xa - xa0) / paso) % 4) * 25}%`;
        e.style.backgroundPositionX = `${(gira ? 4 : Math.floor((xe0 - xe) / paso) % 4) * 25}%`;
        xaU = xa + cw / 2; xeU = xe + cw / 2; avanceU = lin; // ronda 28: la estela se desvanece con el avance
        raiz.classList.toggle('llegaron', lin >= 0.985);
        if (lin < 0.985) abrir(null);
      };
      const pedir = () => { if (visible && !pedido) { pedido = true; requestAnimationFrame(pintar); } };
      new IntersectionObserver((en) => { visible = en[0].isIntersecting; pedir(); }, { rootMargin: '10% 0px' }).observe(paseo);
      addEventListener('scroll', pedir, { passive: true });
      addEventListener('resize', () => { ult = -1; sembrar(); pedir(); });
      pintar();
    }
  }
</script>
