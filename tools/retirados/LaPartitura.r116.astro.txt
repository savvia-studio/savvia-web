    // sección rítmica a su compás (escobilla en cada tiempo, bombo y bajo en el 1 y el 3), muy baja. La música de fondo calla mientras se ve.
    // RONDA 109 · «en la partitura de tu web mantenemos el sonido de la web y ponemos sonidos a las acciones de los agentes»: la música de fondo ya no calla
    // (fuera la sección rítmica propia y las notas de la ronda 100); cada acción suena como lo que es: las cartas que caen se POSAN en la mesa, las
    // ediciones son un clic y un tecleo, abrir el cuaderno es papel, cada parte que escribe un agente es alguien tecleando, cada pieza del hero que se
    // monta un clic, publicar (Fine) es el intro y un timbre suave en el tono de la web, y lo que sale volando, un clic
    const SON: Son[] = [
      ...CASC.map((c) => [c.t, () => posar({ vol: 0.03 + (c.i % 3) * 0.006 })] as Son),
      ...EDIT.map((e) => [b(e.at), () => { clic({ vol: 0.04 }); teclear(4, { vol: 0.028, ritmo: 0.06 }); }] as Son),
      [b(ABRE), () => papel({ vol: 0.07, dur: 0.38 })],
      ...PARTE.map((p) => [b(p.a), () => teclear(Math.round((p.z - p.a) * 10), { vol: 0.026, ritmo: 0.055 })] as Son),
      ...Object.values(PIEZA).map((p) => [b(p.at), () => clic({ vol: 0.035 })] as Son),
      [b(FINE), () => { tecla({ vol: 0.05, fuerte: true }); rhodesSuelto(84, 0.022, 1.4, 0.08, false); rhodesSuelto(91, 0.014, 1.6, 0.14, false); }],
      ...CONS.map((c) => [c.A + 0.1, () => clic({ vol: 0.025 })] as Son),
    ];
    const sonar = (t0: number, t1: number) => {
      for (const [ts, fn] of SON) if ((t1 >= t0 && ts > t0 && ts <= t1) || (t1 < t0 && (ts > t0 || ts <= t1))) fn();
    };

    // ── PINTAR el fotograma t (función pura del tiempo)
    const pintar = (t: number) => {
      const v = camara(t), K = k * v.S;
      pon(cam, '@transform', `translate(${f2(k * (W / 2 - v.S * v.cx))} ${f2(k * (H / 2 - v.S * v.cy))}) scale(${K.toFixed(5)})`);
      pon(cam, '--u', (1 / K).toFixed(5));
      const tb = t / BEAT;
      const fin = sm(tb, 19.4, 19.95);

      for (const r of cajas) {
        let off = 0;
        for (const p of PULSOS) {
          if (p.lado !== r.lado || p.i !== r.i) continue;
          const tp = r.lado === 'in' ? p.t0 : p.t0 + p.d, e = t - tp;
          if (e > 0 && e < 1.6) off += (r.lado === 'in' ? 6 : 5) * Math.sin(e * 44) * Math.exp(-e * 3.4);
        }
        const P = r.P.slice();
        if (modo === 'h') { P[3] += off; P[5] += off * 0.6; } else { P[2] += off; P[4] += off * 0.6; }
        const d = off ? dDe(P) : dDe(r.P);
        pon(r.cuerda, '@d', d);
        let ver = 1, dib = 1;
        if (r.lado === 'out') {
          const cn = CONS.find((x) => x.o === r.i)!, p0 = cn.A - 0.35;
          if (t >= b(ZAMB[1]) && t < p0) { ver = 0; dib = 0; }
          else if (t >= p0 && t < cn.A + 0.1) { dib = EZ.flujo(prog(t, p0, cn.A + 0.1)); ver = 0; }
          if (t >= cn.A + 0.1) ver = EZ.sale(prog(t, cn.A + 0.1, cn.A + 0.6));
        }
        pon(r.cuerda, 'stroke-dasharray', dib >= 1 ? 'none' : `${f3(dib)} 2`);
        let act: Pulso | null = null;
        for (const p of PULSOS) if (p.lado === r.lado && p.i === r.i && t >= p.t0 && t <= p.t0 + p.d + 0.08) act = p;
        if (act) {
          const h = EZ.flujo(prog(t, act.t0, act.t0 + act.d)), seg = 0.16;
          pon(r.pulso, '@d', d);
          pon(r.pulso, 'stroke', AG[act.ag]);
          pon(r.pulso, 'stroke-dasharray', `${f3(Math.min(seg, h))} 2`);
          pon(r.pulso, 'stroke-dashoffset', f3(-(h - Math.min(seg, h))));
          pon(r.pulso, 'opacity', f3(1 - prog(t, act.t0 + act.d, act.t0 + act.d + 0.08)));
          const [x, y] = bez(P, uDe(r.tabla, h));
          pon(r.cabeza, '@transform', `translate(${f2(x)} ${f2(y)})`);
          pon(r.cabeza, 'fill', AG[act.ag]);
          pon(r.cabeza, 'stroke', AG[act.ag]);
          pon(r.cabeza, 'opacity', f3(Math.min(1, prog(t, act.t0, act.t0 + 0.06)) * (1 - prog(t, act.t0 + act.d - 0.04, act.t0 + act.d + 0.04))));
        } else {
          pon(r.pulso, 'opacity', '0');
          pon(r.cabeza, 'opacity', '0');
        }
        // la caja: se enciende con el color de quien la toca (cascada, edición, construcción) o queda seleccionada
        let luz = 0, ag = 0;
        const sube = (x: number, a: number) => { if (x > luz) { luz = x; ag = a; } };
        if (r.lado === 'in') {
          for (const q of CASC) if (q.i === r.i) sube(blip(t, q.t, 0.7) * 0.9, q.ag);
          for (const e of EDIT) if (e.i === r.i) sube(sm(t, b(e.at), b(e.at) + 0.12) * (1 - sm(t, b(SEL_FIN), b(SEL_FIN) + 0.3)), e.ag);
          for (const cn of CONS) if (cn.i === r.i) sube(blip(t, cn.A - 0.85, 0.9), cn.ag);
        } else {
          const cn = CONS.find((x) => x.o === r.i)!;
          sube(blip(t, cn.A + 0.1, 0.9), cn.ag);
          if (cn.o === 3) sube(sm(t, cn.A + 0.1, cn.A + 0.25) * (1 - fin), cn.ag);
        }
        pon(r.g, '--c', AG[ag]);
        pon(r.g, '--l', f3(luz));
        const pop = r.lado === 'out' ? lerp(0.9, 1, ver) : 1;
        pon(r.g, 'opacity', f3(ver));
        pon(r.g, '@transform', `translate(${f2(r.x + (r.w * (1 - pop)) / 2)} ${f2(r.y + (r.h * (1 - pop)) / 2)}) scale(${f3(pop)})`);
      }

      let fl = 0, fa = 0;
      for (const g of GOLPES) { const x = blip(t, g.t, 0.6); if (x > fl) { fl = x; fa = g.ag; } }
      pon(filo, 'stroke', `color-mix(in oklab, ${AG[fa]} ${Math.round(fl * 100)}%, var(--filo-reposo))`);
      pon(filo, '--g', f3(fl));
      const vT = 1 - clamp((Math.log(v.S) - Math.log(FOLD.S * 1.15)) / (Math.log(FOLD.S * 2.6) - Math.log(FOLD.S * 1.15)));
      pon(gTexto, 'opacity', f3(vT));
      GOLPES.forEach((g, q) => pon(notas[q], 'opacity', f3(sm(t, g.t - 0.02, g.t + 0.14) * (1 - fin))));

      // el cuaderno: nace del propio papel de la hoja según la cámara se acerca
      const wo = clamp((Math.log(v.S) - Math.log(8)) / (Math.log(24) - Math.log(8)));
      const w = ventanaEn(v);
      pon(win, 'visibility', wo > 0.001 ? 'visible' : 'hidden');
      pon(win, 'opacity', f3(EZ.fundido(wo)));
      if (wo > 0.001) {
        pon(win, 'transform', `translate(${f2(w.x * k)}px, ${f2(w.y * k)}px) scale(${(w.s * k).toFixed(5)})`);
        pintarCuaderno(tb);
      }

      curs.forEach((c, q) => {
        const [x, y] = manoEn(q, t);
        pon(c, 'transform', `translate3d(${f2(x * k)}px, ${f2(y * k)}px, 0)`);
        let cl = 0, on = 0;
        for (const tc of CLICS[q]) { cl = Math.max(cl, blip(tb, tc, 0.5)); const p = prog(tb, tc, tc + 0.7); if (p > 0 && p < 1) on = p; }
        pon(c, '--clic', f3(cl));
        pon(c, '--onda', f3(on));
      });
    };

    // ── RONDA 94 · LA CINTA (sf-card de la web real): cada tarjeta llega a su sitio con la cinta subiendo, nítida; la anterior se queda arriba al 28 %,
    // desenfocada y a ×0,95; llega en esqueleto, el esqueleto se funde en la cabecera y las filas entran escalonadas (suben 6 px y crecen desde ×0,97)
    const SF_A = [PIEZA.card.at, 12.15, 13.85], SF_VIAJE = 0.75;
    const pintarCinta = (tb: number) => {
      const yF = rectLocal(feed).y, alto = feed.parentElement!.offsetHeight, ancla = alto * 0.2;
      const ys = sf.map((c) => rectLocal(c.e).y - yF);
      // posición de la cinta: entra desde abajo con la primera y sube una tarjeta en cada llegada
      let y = ys[0] + (1 - EZ.flujo(prog(tb, SF_A[0], SF_A[0] + SF_VIAJE))) * -alto * 0.55;
      for (let k = 1; k < sf.length; k++) y += (ys[k] - ys[k - 1]) * EZ.flujo(prog(tb, SF_A[k], SF_A[k] + SF_VIAJE));
      pon(feed, 'transform', `translateY(${f2(ancla - y)}px)`);
      sf.forEach((c, k) => {
        const a = SF_A[k], sig = SF_A[k + 1] ?? 99;
        const v = EZ.fundido(prog(tb, a, a + SF_VIAJE)) * (1 - EZ.fundido(prog(tb, sig, sig + SF_VIAJE)));
        pon(c.e, 'opacity', f3(0.28 + 0.72 * v));
        pon(c.e, 'filter', v > 0.995 ? 'none' : `blur(${f2(5 * (1 - v))}px)`);
        pon(c.e, 'transform', `scale(${(0.95 + 0.05 * v).toFixed(4)})`);
        const h = sm(tb, a + 0.45, a + 0.75);
        pon(c.skel, 'opacity', f3(1 - h));
        pon(c.head, 'opacity', f3(h));
        const f0 = a + (c.pop ? 1.0 : 0.62), paso = c.pop ? 0.26 : 0.2;
        c.filas.forEach((e, j) => pon(e, '--p', f3(EZ.sale(prog(tb, f0 + j * paso, f0 + j * paso + 0.45)))));
        if (c.pop) pon(c.pop, '--p', f3(EZ.sale(prog(tb, a + 0.78, a + 1.15))));
        if (c.hl) pon(c.hl, 'opacity', f3(sm(tb, a + 1.45, a + 1.75)));
      });
    };

    const pintarCuaderno = (tb: number) => {
      sistemas.forEach((e, q) => {
        const p = PARTE[q];
        let d = 0;
        for (const x of Object.values(PIEZA)) if (x.de === q) d = Math.max(d, blip(tb, x.at, 0.9));
        pon(e, '--p', f3(prog(tb, p.a, p.z)));
        pon(e, '--o', f3(sm(tb, p.a - 0.05, p.a + 0.15)));
        pon(e, '--c', AG[p.ag]);
        pon(e, '--d', f3(d));
      });
      // RONDA 94 · en la web real el hero entra con un FUNDIDO (sin palabras que suban): cada pieza, al ponerla su agente, se funde igual
      for (const [n, x] of Object.entries(PIEZA)) pon(pz[n], '--p', f3(EZ.fundido(prog(tb, x.at, x.at + (n === 'fondo' ? 1.1 : 0.6)))));
      // el punto del eyebrow: vaivén senoidal (real 2,4 s; aquí 4 tiempos = 2,6 s, cabe cinco veces en el bucle)
      pon(punto, '--late', f3(0.5 - 0.5 * Math.cos((2 * Math.PI * tb) / 4)));
      pintarCinta(tb);
      pon(doble, '--p', f3(EZ.sale(prog(tb, DOBLE, DOBLE + 0.3))));
      pon(doble, '--f', f3(sm(tb, FINE, FINE + 0.3)));
    };

    // ── el reloj
    let t0 = 0, raf = 0, jugando = false, tPrev = -1, tQuieto = 0, quietoDesde = 0, buscado: number | null = null;
    const tAhora = () => (((performance.now() - t0) / 1000) % LOOP + LOOP) % LOOP;
    const tick = () => {
      raf = 0;
      if (!jugando || buscado !== null) return;
      const t = tAhora();
      if (tPrev >= 0 && ((t - tPrev + LOOP) % LOOP) < 0.25) sonar(tPrev, t);
      tPrev = t;
      pintar(t);
      raf = requestAnimationFrame(tick);
    };

    montar();
    pintar(0);
    const rehacer = () => { montar(); pintar(buscado ?? (jugando ? tAhora() : 0)); };
    mqV.addEventListener('change', rehacer);
    let anchoPrevio = cine.clientWidth;
    new ResizeObserver(() => { if (Math.abs(cine.clientWidth - anchoPrevio) > 1) { anchoPrevio = cine.clientWidth; rehacer(); } }).observe(cine);
    document.fonts?.ready.then(rehacer);

    (window as any).__partitura = {
      LOOP, BEAT,
      ir: (t: number) => { buscado = t; pintar(t); },
      seguir: () => { buscado = null; t0 = performance.now(); tPrev = -1; if (jugando && !raf) raf = requestAnimationFrame(tick); },
      t: () => (buscado ?? tAhora()),
    };

    if (reducido) { sec.classList.add('quieta'); return; }
    new IntersectionObserver((e) => {
      const ver = e[0].isIntersecting;
      if (ver && !jugando) {
        jugando = true;
        t0 = performance.now() - (performance.now() - quietoDesde > 2000 ? 0 : tQuieto * 1000);
        tPrev = -1;
        if (!raf) raf = requestAnimationFrame(tick);
      } else if (!ver && jugando) {
        jugando = false;
        tQuieto = tAhora();
        quietoDesde = performance.now();
      }
    }, { rootMargin: '0px 0px -12% 0px' }).observe(cine);
  }
</script>
