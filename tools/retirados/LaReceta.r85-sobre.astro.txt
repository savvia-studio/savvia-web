// LA PARTITURA · RONDA 85. Andoni: «el ticket tiene que parecer una entrada de concierto de verdad, estilo clásico y que parezca premium, como algo muy hecho a
// medida; e igual quitamos el sonido de que se imprime: buscamos otro concepto». Antes (rondas 53–84): la impresora térmica y su ticket
// (tools/retirados/LaReceta.r84-impresora.astro.txt).
// CONCEPTO: la entrada llega en un SOBRE LACRADO (el lacre lleva el isotipo de savvia). Con el scroll: el lacre se rompe, la solapa se abre y la ENTRADA sale del
// sobre; después se recorre su PROGRAMA, movimiento a movimiento (cada uno suena y, a su lado, aparece su pieza digital: lo que de verdad hace esa parte de
// la partitura en la web de Neety); al final la MATRIZ se rasga por el troquel y cae el sello «admitido». Queda la entrada y el botón a la librería.
// La entrada: papel verjurado crema, tinta berenjena de Neety, filete doble con una GUILLOCHE (la del papel moneda, generada aquí) y detalles en su naranja
// como si fueran estampados en caliente; serif de titulares, versalitas espaciadas en mono. Móvil: vertical, con la matriz abajo. Sin movimiento: todo quieto.
import { receta } from '../data/home';
import isoRaw from '../../public/savvia-isotipo.svg?raw';
const ISO = isoRaw.match(/ d="([^"]+)"/)![1];
const E = receta.entrada;
const TOKEN = /(#[0-9A-Fa-f]{6}|\b[\w/-]+\.md\b)/g;
// la GUILLOCHE: dos senoides entrelazadas en una tira (se repite a lo largo de la banda) y una ROSETA (curvas polares superpuestas) para la matriz
const f = (n: number) => n.toFixed(2);
const tira = (() => {
  const W = 48, H = 12, ps: string[] = [];
  for (const [fase, amp] of [[0, 4.6], [Math.PI, 4.6], [Math.PI / 2, 2.4], [(3 * Math.PI) / 2, 2.4]]) {
    let d = ''; for (let x = 0; x <= W; x += 1) d += (x ? 'L' : 'M') + f(x) + ' ' + f(H / 2 + amp * Math.sin((x / W) * Math.PI * 4 + fase));
    ps.push(`<path d='${d}' fill='none' stroke='%23431b44' stroke-width='.45' opacity='.75'/>`);
  }
  return `url("data:image/svg+xml;utf8,<svg xmlns='http://www.w3.org/2000/svg' width='${W}' height='${H}' viewBox='0 0 ${W} ${H}'>${ps.join('')}</svg>")`;
})();
const roseta = (() => {
  const out: string[] = [];
  for (let c = 0; c < 6; c++) {
    let d = '';
    for (let i = 0; i <= 360; i++) { const th = (i / 360) * Math.PI * 2, r = 30 + 9 * Math.sin(th * 12 + c * 0.52) + 3 * Math.sin(th * 36); d += (i ? 'L' : 'M') + f(50 + r * Math.cos(th)) + ' ' + f(50 + r * Math.sin(th)); }
    out.push(`<path d="${d}Z" />`);
  }
  return out.join('');
})();
const romano = (m: string) => (m.includes('.') ? m.split('.')[0] : '');
const tempo = (m: string) => (m.includes('.') ? m.split('.')[1].trim() : m);
const cap = (t: string) => t.charAt(0).toUpperCase() + t.slice(1);

<script>
  import { tocar, pizzicato, papel } from '../scripts/musica';
  const sec = document.querySelector<HTMLElement>('[data-receta]');
  if (sec) {
    const reducido = matchMedia('(prefers-reduced-motion: reduce)').matches;
    const escena = sec.querySelector<HTMLElement>('[data-sala-entrada]')!, mesa = sec.querySelector<HTMLElement>('[data-mesa]')!;
    const movs = [...sec.querySelectorAll<HTMLElement>('[data-mov]')], digs = [...sec.querySelectorAll<HTMLElement>('[data-dig]')];
    const N = movs.length;
    // la melodía del programa: una nota por movimiento (sube y resuelve)
    const MELODIA = [62, 66, 69, 71, 74];
    if (reducido) { mesa.classList.add('quieta', 'abierta', 'fuera-sobre', 'rasgada', 'sellada'); movs.forEach((m) => m.classList.add('visto')); digs[N - 1].classList.add('on'); }
    else {
      let estado = { abierta: false, fuera: false, activo: -1, rasgada: false, sellada: false }, raf = 0;
      const clamp = (x: number) => Math.min(1, Math.max(0, x));
      const pintar = () => {
        raf = 0;
        const r = escena.getBoundingClientRect(), recorrido = r.height - innerHeight;
        const p = clamp(-r.top / Math.max(1, recorrido));
        mesa.style.setProperty('--p', p.toFixed(4));
        // 1 · el lacre se rompe y la solapa se abre
        const abrir = clamp((p - 0.03) / 0.09);
        mesa.style.setProperty('--abrir', abrir.toFixed(3));
        if (abrir > 0.05 && !estado.abierta) { estado.abierta = true; mesa.classList.add('abierta'); pizzicato(57, { vol: 0.09 }); papel({ vol: 0.05, dur: 0.18 }); }
        if (abrir <= 0.05 && estado.abierta) { estado.abierta = false; mesa.classList.remove('abierta'); }
        // 2 · la entrada sale del sobre y el sobre se va
        const sale = clamp((p - 0.1) / 0.14), seVa = clamp((p - 0.2) / 0.1);
        mesa.style.setProperty('--sale', sale.toFixed(3)); mesa.style.setProperty('--seva', seVa.toFixed(3));
        const fuera = seVa > 0.99; if (fuera !== estado.fuera) { estado.fuera = fuera; mesa.classList.toggle('fuera-sobre', fuera); }
        // 3 · el programa, movimiento a movimiento (y su pieza al lado)
        const k = p < 0.32 ? -1 : Math.min(N - 1, Math.floor((p - 0.32) / (0.52 / N)));
        if (k !== estado.activo) {
          estado.activo = k;
          movs.forEach((m, i) => { m.classList.toggle('on', i === k); m.classList.toggle('visto', i < k); });
          digs.forEach((d, i) => d.classList.toggle('on', i === k));
          if (k >= 0) tocar(MELODIA[k], { vol: 0.08, dur: 2 });
        }
        // 4 · la matriz se rasga por el troquel; 5 · cae el sello
        const rasgada = p > 0.86, sellada = p > 0.91;
        if (rasgada !== estado.rasgada) { estado.rasgada = rasgada; mesa.classList.toggle('rasgada', rasgada); if (rasgada) papel(); }
        if (sellada !== estado.sellada) { estado.sellada = sellada; mesa.classList.toggle('sellada', sellada); if (sellada) { tocar(50, { vol: 0.1, dur: 2.4 }); tocar(62, { vol: 0.06, dur: 2.4 }); } }
      };
      const pedir = () => { if (!raf) raf = requestAnimationFrame(pintar); };
      let cerca = false;
      new IntersectionObserver((e) => { cerca = e[0].isIntersecting; if (cerca) pedir(); }, { rootMargin: '20% 0px' }).observe(escena);
      addEventListener('scroll', () => cerca && pedir(), { passive: true });
      addEventListener('resize', pedir);
      pintar();
    }
    (window as any).__partitura = () => ({ clases: mesa.className, activo: movs.findIndex((m) => m.classList.contains('on')), p: mesa.style.getPropertyValue('--p') });
  }
</script>
